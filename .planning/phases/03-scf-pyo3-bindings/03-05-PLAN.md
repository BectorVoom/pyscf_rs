<!-- DF-HF J/K formula (upstream pyscf/df/df_jk.py:31-148) -->
B is the n_aux × nao × nao tensor satisfying:
  ∫∫ (μν|λσ) ≈ Σ_Q B^Q_{μν} · B^Q_{λσ}

J[μν] = Σ_λσ (μν|λσ) D[λσ]
     ≈ Σ_λσ Σ_Q B^Q_{μν} · B^Q_{λσ} D[λσ]
     = Σ_Q B^Q_{μν} · ρ^Q          where ρ^Q = Σ_λσ B^Q_{λσ} D[λσ]

K[μν] = Σ_λσ (μλ|νσ) D[λσ]
     ≈ Σ_λσ Σ_Q B^Q_{μλ} · B^Q_{νσ} D[λσ]
     = Σ_Q (B^Q D)_{μλ} · B^Q_{νσ}   — needs intermediate tensor (Wabef-like)

       fn evaluate_int3c2e_with_auxmol(mol: &Mole, auxmol: &Mole)
           -> Result<IntorOutput, PyscfRsError>
       {
           // Both Moles must be built.
           if !mol._built || !auxmol._built {
               return Err(PyscfRsError::Core(
                   pyscf_core::CoreError::InvalidMolecule(
                       "intor_with_auxmol: both mol and auxmol must be built".into()
                   )
               ));
           }
           let nao    = mol.nao_nr;
           let naux   = auxmol.nao_nr;

           // Build merged BasisSet (orbital shells first, then aux shells) using
           // cintx_core::BasisSet primitives. Phase 2 D-03 already exposes
           // `Mole::cintx_basis() -> Arc<BasisSet>`; here we need a temporary
           // merged set that doesn't outlive this call.
           let orb = mol.cintx_basis()?;
           let aux = auxmol.cintx_basis()?;
           let merged = cintx_core::BasisSet::concat(&orb, &aux).map_err(|e| {
               PyscfRsError::Core(pyscf_core::CoreError::InvalidMolecule(format!(
                   "intor_with_auxmol: BasisSet::concat failed: {}", e
               )))
           })?;

           // Resolve operator id for int3c2e_sph via cintx-ops Resolver.
           let descriptor = Resolver::descriptor_by_symbol("int3c2e_sph").map_err(|e| {
               PyscfRsError::Core(pyscf_core::CoreError::InvalidMolecule(format!(
                   "cintx-ops resolver does not know 'int3c2e_sph': {}", e
               )))
           })?;

           // Iterate (i, j, k) shell triples and stitch into a (nao, nao, naux)
           // F-order output. Detailed body parallels `evaluate_arity2` in this
           // file — executor adapts to the cintx-rs arity-3 SessionRequest shape.
           // For Phase 3 minimum viable, a straightforward triple loop is
           // sufficient (test corpus has bounded nao/naux).
           let mut out = vec![0.0f64; nao * nao * naux];
           // … cintx-rs arity-3 query — see Phase 2 plan 02-09 verification rollup
           // for the proven invocation pattern; adapt the shls_slice argument to
           // restrict the third index to the aux block.

           Ok(IntorOutput {
               values: out,
               shape: vec![nao, nao, naux],
               layout: IntorLayout::ScalarFOrder,
           })
       }

       fn evaluate_int2c2e_aux(auxmol: &Mole) -> Result<IntorOutput, PyscfRsError> {
           // Plain arity-2 intor on auxmol alone — reuse the existing `intor` path:
           intor(auxmol, "int2c2e_sph")
       }
       ```

       //! cholesky_eri: build the DF B-integrals.
       //!
       //! Source: pyscf/df/incore.py:cholesky_eri (verbatim algorithm port).
       //! Algorithm:
       //!   1. Build auxmol (Mole-like object with the aux basis)
       //!   2. Compute (μν|P) 3-center integrals via pyscf_gto::intor_with_auxmol
       //!      (Phase 3 plan 03-05 Task 0 — added because Phase 2 dispatcher
       //!      lacked an auxmol overload; see checker iteration 1 WARNING 5 fix)
       //!   3. Compute (P|Q) 2-center integrals via pyscf_gto::intor_with_auxmol
       //!      ("int2c2e_sph" on auxmol)
       //!   4. Cholesky-decompose (P|Q) = L @ L.T via pyscf-algebra::cholesky
       //!   5. Solve L @ B = int3c → B_{μν,Q} (forward triangular solve)
       //!
       //! Phase 3 limitation: in-memory only (D-11). Phase 6 adds HDF5 spill.
       use pyscf_core::{Mole, PyscfRsError};
       use crate::error::DfError;

       pub struct DfIntegrals {
           /// B-integrals: flattened layout `b_uvq[mu*nao*naux + nu*naux + q]` (row-major).
           /// (Phase 3 ships row-major; if profiling shows F-order is better for the
           /// J/K contraction kernels, plan 03-10's optimization round can switch.)
           pub b_uvq: Vec<f64>,
           pub naux:  usize,
           pub nao:   usize,
       }

       pub fn cholesky_eri(mol: &Mole, auxbasis: &str) -> Result<DfIntegrals, PyscfRsError> {
           let nao = mol.nao_nr;
           // Step 1: build auxmol (copy of mol with basis replaced by auxbasis).
           let mut auxmol = mol.clone();
           auxmol.basis = pyscf_core::ParsedBasis::Name(auxbasis.to_string());
           auxmol.build()?;
           let naux = auxmol.nao_nr;

           // Step 2: 3-center integrals — NOW via pyscf_gto::intor_with_auxmol
           // (Task 0 of this plan). Returns IntorOutput shape (nao, nao, naux),
           // F-order layout.
           let int3c = pyscf_gto::intor_with_auxmol(mol, "int3c2e_sph", &auxmol)?;
           // Step 3: 2-center (P|Q) — via the same wrapper, dispatched to plain intor.
           let int2c = pyscf_gto::intor_with_auxmol(mol, "int2c2e_sph", &auxmol)?;
           // Step 4: Cholesky of (P|Q) = L @ L.T. pyscf-algebra::cholesky returns L.
           let l = pyscf_algebra::cholesky(&int2c.values, naux)
               .map_err(|e| DfError::Algebra(e))
               .map_err(|e| PyscfRsError::Core(pyscf_core::CoreError::Other(format!("{}", e))))?;
           // Step 5: forward triangular solve L @ B = int3c → B.
           let mut b_uvq = vec![0.0_f64; nao * nao * naux];
           for mu in 0..nao {
               for nu in 0..nao {
                   let mut rhs = vec![0.0_f64; naux];
                   for q in 0..naux {
                       // int3c is F-order [nao, nao, naux]: offset = mu + nu*nao + q*nao*nao
                       rhs[q] = int3c.values[mu + nu * nao + q * nao * nao];
                   }
                   let b = forward_substitute(&l, &rhs, naux);
                   for q in 0..naux {
                       b_uvq[mu * nao * naux + nu * naux + q] = b[q];
                   }
               }
           }
           Ok(DfIntegrals { b_uvq, naux, nao })
       }

       /// Forward-substitute: solve L · x = b for lower-triangular L stored row-major.
       fn forward_substitute(l: &[f64], b: &[f64], n: usize) -> Vec<f64> {
           let mut x = vec![0.0_f64; n];
           for i in 0..n {
               let mut s = b[i];
               for j in 0..i {
                   s -= l[i * n + j] * x[j];
               }
               x[i] = s / l[i * n + i];
           }
           x
       }
       ```

       ```rust
       //! Density-fitted J/K builder.
       //!
       //! Source: pyscf/df/df_jk.py:31-148 — get_jk reference.
       //! Algorithm:
       //!   J[μν] = Σ_Q B^Q_{μν} · ρ^Q   where ρ^Q = Σ_λσ B^Q_{λσ} D[λσ]
       //!   K[μν] = Σ_Q (B^Q D)_{μλ} · B^Q_{νλ} — needs intermediate W^Q_{μλ}
       use pyscf_core::{Density, PyscfRsError};
       use crate::cholesky_eri::DfIntegrals;

       pub fn get_jk_df(
           dm: &Density,
           df: &DfIntegrals,
       ) -> Result<(Density, Density), PyscfRsError> {
           let nao = df.nao;
           let naux = df.naux;
           assert_eq!(dm.nao, nao, "density nao={} does not match DF nao={}", dm.nao, nao);

           // ρ^Q = Σ_λσ B^Q_{λσ} D[λσ]   — naux-element vector
           let mut rho_q = vec![0.0_f64; naux];
           for q in 0..naux {
               let s = pyscf_algebra::oracle_sum(
                   (0..nao).flat_map(|lambda| (0..nao).map(move |sigma| {
                       df.b_uvq[lambda * nao * naux + sigma * naux + q]
                           * dm.data[lambda * nao + sigma]
                   }))
               );
               rho_q[q] = s;
           }

           // J[μν] = Σ_Q B^Q_{μν} · ρ^Q
           let mut j = vec![0.0_f64; nao * nao];
           for mu in 0..nao {
               for nu in 0..nao {
                   j[mu * nao + nu] = pyscf_algebra::oracle_sum(
                       (0..naux).map(|q| df.b_uvq[mu * nao * naux + nu * naux + q] * rho_q[q])
                   );
               }
           }

           // K[μν] = Σ_Q (B^Q D)_{μλ} · B^Q_{νλ}
           let mut w = vec![0.0_f64; naux * nao * nao];
           for q in 0..naux {
               for mu in 0..nao {
                   for lambda in 0..nao {
                       w[q * nao * nao + mu * nao + lambda] = pyscf_algebra::oracle_sum(
                           (0..nao).map(|sigma|
                               df.b_uvq[mu * nao * naux + sigma * naux + q]
                                   * dm.data[sigma * nao + lambda]
                           )
                       );
                   }
               }
           }
           let mut k = vec![0.0_f64; nao * nao];
           for mu in 0..nao {
               for nu in 0..nao {
                   k[mu * nao + nu] = pyscf_algebra::oracle_sum(
                       (0..naux).flat_map(|q| (0..nao).map(move |lambda| {
                           w[q * nao * nao + mu * nao + lambda]
                               * df.b_uvq[nu * nao * naux + lambda * naux + q]
                       }))
                   );
               }
           }

           Ok((
               Density { nao, data: j },
               Density { nao, data: k },
           ))
       }
       ```
