<!-- pyscf-py build.rs — feature mutex guard (checker iteration 1 WARNING 4) -->
```rust
// crates/pyscf-py/build.rs
//
// Cargo features are not natively mutually exclusive. Without this guard a
// developer or CI run can pass `--features abi3-py310,free-threading` and
// produce a wheel whose C ABI is undefined: abi3 promises a stable
// limited-API surface, but the free-threaded interpreter (3.13t) drops the
// GIL semantics that abi3 callers rely on.
//
// Source: RESEARCH §"CRITICAL abi3 vs free-threaded ABI conflict" + checker
// iteration 1 WARNING 4.
fn main() {
    let abi3 = std::env::var("CARGO_FEATURE_ABI3_PY310").is_ok();
    let ft   = std::env::var("CARGO_FEATURE_FREE_THREADING").is_ok();
    if abi3 && ft {
        panic!(
            "pyscf-py: features `abi3-py310` and `free-threading` are mutually exclusive \
             — choose one. See .planning/phases/03-scf-pyo3-bindings/03-RESEARCH.md \
             §'CRITICAL abi3 vs free-threaded ABI conflict'"
        );
    }
    println!("cargo:rerun-if-changed=build.rs");
}
```

       //! BIND-06 lint: forbid `lazy_static!` in crates/pyscf-py/.
       //!

       //! free-threaded Python (3.13t), `lazy_static!` deadlocks because it uses

       use anyhow::{Context, Result};
       use std::fs;
       use std::path::Path;
       use std::process::ExitCode;

       fn main() -> Result<ExitCode> {
           let root = std::env::current_dir()
               .context("failed to read CWD")?
               .join("crates/pyscf-py");
           if !root.exists() {
               eprintln!(
                   "check-forbid-lazy-static: pyscf-py crate not yet present — pass"
               );
               return Ok(ExitCode::from(0));
           }
           let hits = scan(&root)?;
           if hits.is_empty() {
               eprintln!(
                   "check-forbid-lazy-static: PASS — no `lazy_static!` in crates/pyscf-py/ (BIND-06)"
               );
               Ok(ExitCode::from(0))
           } else {
               eprintln!(
                   "check-forbid-lazy-static: FAIL — `lazy_static!` invocations found:"
               );
               for h in &hits {
                   eprintln!("  {}", h);
               }
               eprintln!("Use `pyo3::sync::PyOnceLock` instead (BIND-06, free-thread-safe).");
               Ok(ExitCode::from(2))
           }
       }

       fn scan(dir: &Path) -> Result<Vec<String>> {
           let mut hits = vec![];
           for entry in fs::read_dir(dir)? {
               let entry = entry?;
               let path = entry.path();
               if path.is_dir() {
                   if path.file_name().and_then(|s| s.to_str()) == Some("target") {
                       continue;
                   }
                   hits.extend(scan(&path)?);
               } else if path.extension().and_then(|s| s.to_str()) == Some("rs") {
                   let content = fs::read_to_string(&path)
                       .with_context(|| format!("reading {}", path.display()))?;

                       let trimmed = line.trim_start();
                       if trimmed.starts_with("//") {

                       }
                       if line.contains("lazy_static!") {
                           hits.push(format!("{}:{}: {}", path.display(), i + 1, line.trim()));
                       }
                   }
               }
           }
           Ok(hits)
       }
