```rust
    // Ensure F-contiguous: transpose-copy via standard layout pivot.
```

       ```rust
       //! HDF5 chkfile primitives — h5py-compatible reads/writes.
       //!
       //! Source: pyscf/lib/chkfile.py:28-191. Every encoding choice (VL Unicode,
       //! F-order mo_coeff, scalar/1D datasets) matches upstream byte-for-byte
       //! (verified empirically via ORACLE-08 round-trip in plan 03-08).
       use crate::error::ChkfileError;
       use hdf5_metno as hdf5;
       use hdf5_metno::types::VarLenUnicode;
       use ndarray::{Array1, Array2, ArrayView2};
       use std::path::Path;
       use std::str::FromStr;

       /// Open or create a chkfile for write. Mirrors pyscf/lib/chkfile.py:dump
       /// behaviour: if file exists, append; otherwise create.
       pub fn open_for_write<P: AsRef<Path>>(path: P) -> Result<hdf5::File, ChkfileError> {
           let p = path.as_ref();
           if p.exists() {
               hdf5::File::append(p).map_err(ChkfileError::from)
           } else {
               hdf5::File::create(p).map_err(ChkfileError::from)
           }
       }

       /// Open chkfile for read-only access.
       pub fn open_for_read<P: AsRef<Path>>(path: P) -> Result<hdf5::File, ChkfileError> {
           hdf5::File::open(path).map_err(ChkfileError::from)
       }

       /// Read a sub-group (e.g. group "scf" inside a chkfile).
       pub fn read_group(file: &hdf5::File, name: &str) -> Result<hdf5::Group, ChkfileError> {
           file.group(name).map_err(ChkfileError::from)
       }

       /// Write `mol.dumps()` JSON string under `/mol` as VL Unicode.
       /// Source: pyscf/lib/chkfile.py:179-191 save_mol.
       pub fn write_mol(file: &hdf5::File, mol_json: &str) -> Result<(), ChkfileError> {
           let vl = VarLenUnicode::from_str(mol_json)
               .map_err(|_| ChkfileError::InvalidUtf8)?;
           // If /mol exists (append mode), delete + recreate so the write is idempotent.
           if file.link_exists("mol") {
               file.unlink("mol")?;
           }
           file.new_dataset::<VarLenUnicode>()
               .create("mol")?
               .write_scalar(&vl)?;
           Ok(())
       }

       /// Read `/mol` VL Unicode string.
       pub fn read_mol(file: &hdf5::File) -> Result<String, ChkfileError> {
           let vl: VarLenUnicode = file.dataset("mol")?.read_scalar()?;
           Ok(vl.as_str().to_string())
       }

       /// Write a scalar f64 (e.g. e_tot).
       pub fn write_scalar_f64(group: &hdf5::Group, key: &str, value: f64) -> Result<(), ChkfileError> {
           if group.link_exists(key) { group.unlink(key)?; }
           group.new_dataset::<f64>().create(key)?.write_scalar(&value)?;
           Ok(())
       }

       /// Read a scalar f64.
       pub fn read_scalar_f64(group: &hdf5::Group, key: &str) -> Result<f64, ChkfileError> {
           Ok(group.dataset(key)?.read_scalar::<f64>()?)
       }

       /// Write a 1D f64 array (e.g. mo_energy, mo_occ).
       pub fn write_dataset_1d(group: &hdf5::Group, key: &str, data: &[f64]) -> Result<(), ChkfileError> {
           if group.link_exists(key) { group.unlink(key)?; }
           let arr = Array1::from_vec(data.to_vec());
           group.new_dataset::<f64>()
               .shape([arr.len()])
               .create(key)?
               .write(&arr)?;
           Ok(())
       }

       /// Read a 1D f64 array.
       pub fn read_dataset_1d(group: &hdf5::Group, key: &str) -> Result<Vec<f64>, ChkfileError> {
           let arr: Array1<f64> = group.dataset(key)?.read_1d()?;
           Ok(arr.to_vec())
       }

       /// Write a 2D f64 dataset in C-order (default). For density matrices etc.
       pub fn write_dataset_c_order(group: &hdf5::Group, key: &str, data: &Array2<f64>) -> Result<(), ChkfileError> {
           if group.link_exists(key) { group.unlink(key)?; }
           group.new_dataset::<f64>()
               .shape(data.shape())
               .create(key)?
               .write(data)?;
           Ok(())
       }

       /// Write a 2D f64 dataset preserving F-order (column-major). For mo_coeff.
       /// Source: pyscf/scf/chkfile.py:28-42 — upstream writes mf.mo_coeff which is
       /// F-order LAPACK output. Pitfall 8 mitigation.
       pub fn write_dataset_f_order(group: &hdf5::Group, key: &str, data: ArrayView2<f64>) -> Result<(), ChkfileError> {
           if group.link_exists(key) { group.unlink(key)?; }
           // Convert to F-contiguous owned: as_standard_layout returns C-contig;
           // .reversed_axes() swaps strides; .to_owned() materializes — F-order ownership.
           let f_owned = data.as_standard_layout().reversed_axes().to_owned();
           group.new_dataset::<f64>()
               .shape(f_owned.shape())
               .create(key)?
               .write(&f_owned)?;
           Ok(())
       }

       /// Read a 2D f64 dataset (caller handles F-order vs C-order interpretation).
       pub fn read_dataset_2d(group: &hdf5::Group, key: &str) -> Result<Array2<f64>, ChkfileError> {
           Ok(group.dataset(key)?.read_2d()?)
       }
       ```

       ```rust
       fn mo_coeff_f_order_round_trip() {
       ```

       ```rust
       //! pyscf-scf::chkfile — per-method chkfile schema for SCF.
       //!
       //! Source: pyscf/scf/chkfile.py:25-42 dump_scf / load_scf. Schema:
       //!   /mol            (VL Unicode JSON from mol.dumps() — written at file root)
       //!   /scf/e_tot      (scalar f64)
       //!   /scf/mo_energy  (1D f64, length nao)
       //!   /scf/mo_occ     (1D f64, length nao)
       //!   /scf/mo_coeff   (2D f64, F-order, shape [nao, nao])
       //!
       //! Per D-06: per-method module owns its own schema. Phase 4-7 add their own
       //! Checkpointable impls (KsResult, CcsdResult, OptimState).
       use ndarray::Array2;
       use pyscf_chkfile::{Checkpointable, ChkfileError, primitives};
       use pyscf_chkfile::hdf5;
       use pyscf_core::MOCoefficients;
       use crate::ScfResult;

       impl Checkpointable for ScfResult {
           fn dump(&self, scf_group: &hdf5::Group) -> Result<(), ChkfileError> {
               // Write e_tot, mo_energy, mo_occ, mo_coeff under the passed group.
               // The caller (typically RHF.kernel()) creates the group with name "scf"
               // and writes /mol at the file root.
               primitives::write_scalar_f64(scf_group, "e_tot", self.e_tot.0)?;
               primitives::write_dataset_1d(scf_group, "mo_energy", &self.mo_energy)?;
               primitives::write_dataset_1d(scf_group, "mo_occ", &self.mo_occ)?;
               // mo_coeff in F-order — RESEARCH Pattern 6 + Pitfall 8.
               // MOCoefficients.data is already F-order flat (Phase 1 D-shipped layout).
               // We construct an Array2 view over the F-order buffer with reversed_axes
               // so the as_standard_layout() pivot in write_dataset_f_order works
               // correctly.
               let nao = self.mo_coeff.nao;
               let nmo = self.mo_coeff.nmo;
               // ndarray expects a default-stride view; we transpose: build a C-order
               // ArrayView2 over the data as if it were [nmo, nao] C-order, then
               // .reversed_axes() to get the [nao, nmo] F-order view that the F-write
               // primitive expects.
               let c_view = ndarray::ArrayView2::from_shape((nmo, nao), &self.mo_coeff.data)
                   .map_err(|_| ChkfileError::ShapeMismatch {
                       key: "mo_coeff".into(),
                       expected: vec![nao, nmo],
                       actual:   vec![self.mo_coeff.data.len()],
                   })?
                   .reversed_axes();
               primitives::write_dataset_f_order(scf_group, "mo_coeff", c_view)?;
               Ok(())
           }

           fn load(scf_group: &hdf5::Group) -> Result<Self, ChkfileError> {
               let e_tot = pyscf_core::Energy(primitives::read_scalar_f64(scf_group, "e_tot")?);
               let mo_energy = primitives::read_dataset_1d(scf_group, "mo_energy")?;
               let mo_occ    = primitives::read_dataset_1d(scf_group, "mo_occ")?;
               let mo_coeff_mat: Array2<f64> = primitives::read_dataset_2d(scf_group, "mo_coeff")?;
               // Reconstruct MOCoefficients in F-order layout. The on-disk layout is F-order
               // [nao, nmo]; h5py/hdf5-metno read returns the data as if shape=[nao, nmo] C-order
               // — so we transpose into a true F-order Vec<f64>.
               let nao = mo_coeff_mat.shape()[0];
               let nmo = mo_coeff_mat.shape()[1];
               // Reorder the C-read into F-order flat: for (i, j) F-order, the flat
               // index is i + j*nao = the value at (i, j) in the read matrix.
               let mut data = Vec::with_capacity(nao * nmo);
               for j in 0..nmo {
                   for i in 0..nao {
                       data.push(mo_coeff_mat[(i, j)]);
                   }
               }
               Ok(ScfResult {
                   e_tot,
                   mo_coeff: MOCoefficients { nao, nmo, data, energies: mo_energy.clone(), occupations: mo_occ.clone() },
                   mo_energy,
                   mo_occ,
                   converged: true,   // chkfile is only written on converged SCF per upstream convention
                   cycles: 0,         // not stored in upstream schema
               })
           }
       }

       /// Top-level helper: write entire ScfResult + mol JSON to a chkfile at `path`.
       /// Mirrors upstream `pyscf.lib.chkfile.save` + `pyscf.scf.chkfile.dump_scf`.
       pub fn dump_scf_to_file(path: &std::path::Path, mol_json: &str, result: &ScfResult)
           -> Result<(), ChkfileError>
       {
           let f = primitives::open_for_write(path)?;
           primitives::write_mol(&f, mol_json)?;
           // Recreate /scf group if it exists (idempotent dump).
           if f.link_exists("scf") {
               f.unlink("scf")?;
           }
           let scf = f.create_group("scf")?;
           result.dump(&scf)?;
           f.flush()?;
           Ok(())
       }

       /// Read an ScfResult back from a chkfile.
       pub fn load_scf_from_file(path: &std::path::Path) -> Result<ScfResult, ChkfileError> {
           let f = primitives::open_for_read(path)?;
           let scf = primitives::read_group(&f, "scf")?;
           ScfResult::load(&scf)
       }
       ```

       ```rust
                       prior.mo_coeff.nao, nao
                   ),
               });
           }
           // Build density: D[μν] = Σ_i occ_i * C[μ,i] * C[ν,i].   F-order C.
           let mut data = vec![0.0_f64; nao * nao];
           for mu in 0..nao {
               for nu in 0..nao {

                           prior.mo_occ[i]
                               * prior.mo_coeff.data[mu + i * nao]
                               * prior.mo_coeff.data[nu + i * nao]
                       })
                   );
               }
           }
           Ok(Density { nao, data })
       }
       ```
