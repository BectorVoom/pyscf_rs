       //!
       //! CRITICAL: abi3-py310 forbids `#[pyclass(extends=PyException)]` until Python 3.12.
       //! Workaround per RESEARCH §Pattern 5 + xcfun-py errors.rs:31-83:

       /// Convert a pyscf-core Rust error into a PyErr preserving error-chain info.
       /// Triplet (msg, kind, source_chain) is passed positionally so the Python-side
       /// shim can graft `.kind` and `.source_chain` via the args tuple.
       pub fn pyscf_to_py(err: PyscfRsError) -> PyErr {
           let kind = error_kind(&err);
           let msg = format!("{}", err);
           let source_chain = collect_source_chain(&err);
           PyscfRsRuntimeError::new_err((msg, kind, source_chain))
       }

       /// Convert a Python error (raised inside a subclass override) into a
       /// pyscf-core PyscfRsError so the kernel can `?` propagate.
       pub fn py_to_pyscf(err: PyErr) -> PyscfRsError {
           PyscfRsError::Core(pyscf_core::CoreError::Other(format!(
               "python override raised: {}", err
           )))
       }

       fn error_kind(err: &PyscfRsError) -> &'static str {
           match err {
               PyscfRsError::Core(_)              => "Core",
               PyscfRsError::NotYetImplemented{..} => "NotYetImplemented",
               // Add SCF-specific variants here as pyscf-core grows them.
               _ => "Unknown",
           }
       }

       fn collect_source_chain(err: &dyn std::error::Error) -> Vec<String> {
           let mut chain = Vec::new();
           let mut cur: Option<&dyn std::error::Error> = err.source();
           while let Some(e) = cur {
               chain.push(format!("{}", e));
               cur = e.source();
           }
           chain
       }

       //! NumPy boundary discipline (D-04, BIND-04, Pitfall 5).
       //!
       //! Every PyReadonlyArray2 input runs `is_c_contiguous()` / `is_standard_layout()`
       //! and falls back to `to_owned()` if not. The BIND-04 stride-fuzz test (plan
       //! 03-10) calls each entry with `a`, `a.T`, `a[::2]`, `a[:,1:5]` and asserts
       //! identical answers.
       use numpy::{PyArray2, PyArrayMethods, PyReadonlyArray2, PyUntypedArrayMethods};
       use pyo3::prelude::*;
       use pyscf_core::{Density, MOCoefficients};

       /// Convert a NumPy 2D f64 array into a pyscf-core Density (C-order convention).
       /// Source: xcfun-py numpy_io.rs:9-36 verbatim adapted for f64.
       pub fn to_density<'py>(arr: PyReadonlyArray2<'py, f64>) -> PyResult<Density> {
           let shape = arr.shape();
           if shape.len() != 2 || shape[0] != shape[1] {
               return Err(pyo3::exceptions::PyValueError::new_err(
                   format!("density must be square 2D, got {:?}", shape)));
           }
           // BIND-04: standard-layout check + to_owned fallback.
           let data: Vec<f64> = if arr.is_c_contiguous() {
               arr.as_slice()?.to_vec()
           } else {
               // to_owned reallocates as default-order (C-contiguous) ndarray.
               arr.as_array().to_owned().into_raw_vec_and_offset().0
           };
           Ok(Density { nao: shape[0], data })
       }

       /// Convert NumPy 2D f64 array into MOCoefficients (F-order convention).
       pub fn to_mo_coeff<'py>(arr: PyReadonlyArray2<'py, f64>) -> PyResult<MOCoefficients> {
           let shape = arr.shape();
           let (nao, nmo) = (shape[0], shape[1]);
           // For F-order: if input not F-contiguous, transpose-copy.
           let data: Vec<f64> = if is_f_contiguous(&arr) {
               arr.as_slice()?.to_vec()
           } else {
               let view = arr.as_array();
               let mut buf = Vec::with_capacity(nao * nmo);
               for j in 0..nmo {
                   for i in 0..nao {
                       buf.push(view[[i, j]]);
                   }
               }
               buf
           };
           Ok(MOCoefficients { nao, nmo, data, energies: vec![], occupations: vec![] })
       }

       fn is_f_contiguous(arr: &PyReadonlyArray2<'_, f64>) -> bool {
           // F-contiguous: strides == (8, nao*8). PyReadonlyArray2::strides returns bytes.
           let strides = arr.strides();
           let shape = arr.shape();
           strides.len() == 2
               && strides[0] == std::mem::size_of::<f64>() as isize
               && strides[1] == (shape[0] * std::mem::size_of::<f64>()) as isize
       }

       /// Density → C-contiguous NumPy 2D array.
       pub fn density_to_pyarray<'py>(py: Python<'py>, d: &Density)
           -> PyResult<Bound<'py, PyArray2<f64>>>

           -> PyResult<Bound<'py, PyArray2<f64>>>
       {
           // mc.data is F-order flat. Build a transposed C-order ndarray then convert.
           // For the test corpus (nao ~ 24-70) the cost is negligible.
           let (nao, nmo) = (mc.nao, mc.nmo);
           // Build C-order data: c[(i,j) C-order] = mc.data[i + j*nao] (F-order extract)
           let mut c_data = Vec::with_capacity(nao * nmo);
           for i in 0..nao {
               for j in 0..nmo {
                   c_data.push(mc.data[i + j * nao]);
               }
           }
           let arr = ndarray::Array2::from_shape_vec((nao, nmo), c_data)
               .map_err(|e| pyo3::exceptions::PyValueError::new_err(e.to_string()))?;
           Ok(PyArray2::from_owned_array(py, arr))
       }
