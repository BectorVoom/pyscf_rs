       ```rust
       //! C-DIIS body — Pulay extrapolation (CDIIS).
       //!
       //! Source: pyscf/scf/diis.py:40-87 + Pulay 1980 (DOI:10.1016/0009-2614(80)80396-4).
       //! Verbatim algorithm port. Reductions via oracle_dot/oracle_sum.
       use crate::{DiisError, DiisStorable};

       pub struct Diis<S: DiisStorable + Clone> {
           /// Maximum subspace size (default 8 per pyscf/scf/hf.py:1701).
           pub space:       usize,
           /// Cycle index at which to start extrapolating (default 1 per hf.py:1704).
           pub start_cycle: usize,
           /// Ring-buffer of past iterates.
           bookkeep:        Vec<S>,
           /// Ring-buffer of past error vectors.
           error_vecs:      Vec<Vec<f64>>,
           /// Ring-buffer head (oldest index).
           head:            usize,
       }

       impl<S: DiisStorable + Clone> Diis<S> {
           pub fn new(space: usize) -> Self {
               assert!(space > 0, "DIIS subspace must be > 0");
               Self {
                   space,
                   start_cycle: 1,
                   bookkeep:   Vec::with_capacity(space),
                   error_vecs: Vec::with_capacity(space),
                   head:       0,
               }
           }

           /// Push (current iterate, error vector). Maintains a `space`-sized ring buffer.
           fn push(&mut self, current: S, error: Vec<f64>) {
               if self.bookkeep.len() < self.space {
                   self.bookkeep.push(current);
                   self.error_vecs.push(error);
               } else {
                   self.bookkeep[self.head] = current;
                   self.error_vecs[self.head] = error;
                   self.head = (self.head + 1) % self.space;
               }
           }

           /// Extrapolated next iterate (Pulay).
           /// Source: pyscf/scf/diis.py:48-58 (update method) — Pulay's Lagrange-multiplier
           /// system. B[i,j] = oracle_dot(err_i, err_j); RHS = [0,0,...,0,-1]; sol = LU.
           pub fn extrapolate(&mut self, current: S, error: Vec<f64>) -> Result<S, DiisError> {
               self.push(current, error);
               let n = self.bookkeep.len();
               let dim = n + 1;
               // Build B-matrix in row-major. B[i,j] = oracle_dot(err_i, err_j) for i,j<n.
               let mut b = vec![0.0_f64; dim * dim];
               for i in 0..n {
                   for j in 0..n {
                       b[i * dim + j] = pyscf_algebra::oracle_dot(
                           &self.error_vecs[i], &self.error_vecs[j]
                       );
                   }
                   b[i * dim + n] = -1.0;
                   b[n * dim + i] = -1.0;
               }
               b[n * dim + n] = 0.0;
               // RHS = [0, 0, ..., 0, -1].
               let mut rhs = vec![0.0_f64; dim];
               rhs[n] = -1.0;
               // Solve via pyscf-algebra::solve_linear (plan 03-01 — host-faer LU).
               let c = pyscf_algebra::solve_linear(&b, &rhs, dim)
                   .map_err(|e| match e {
                       pyscf_algebra::AlgebraError::Singular => DiisError::Singular,
                       other => DiisError::Algebra(other),
                   })?;
               // Extrapolated iterate: F_new[k] = Σ_{i<n} c[i] * bookkeep[i].as_flat()[k].
               let flat_len = self.bookkeep[0].len();
               let mut extrap_flat = vec![0.0_f64; flat_len];
               for k in 0..flat_len {
                   // oracle_sum the contributions to preserve Pitfall 9 invariance.
                   extrap_flat[k] = pyscf_algebra::oracle_sum(
                       (0..n).map(|i| c[i] * self.bookkeep[i].as_flat()[k])
                   );
               }
               let mut extrap = self.bookkeep[0].clone();
               extrap.from_flat(&extrap_flat);
               Ok(extrap)
           }
       }

       /// SDF - FDS error vector for SCF use (upstream pyscf/scf/diis.py:68-87).
       /// Reuses pyscf-algebra::gemm for the matrix chain (which itself respects the
       /// algebra wall — pyscf-diis doesn't touch cubecl).
       ///
       /// Inputs: s, d, f are nao × nao C-order Density buffers.
       /// Output: nao*nao C-order flat error vector.
       pub fn err_vec_scf(s: &[f64], d: &[f64], f: &[f64], nao: usize) -> Vec<f64> {
           // sd = S @ D
           let sd = matmul_c(s, d, nao);
           // sdf = sd @ F
           let sdf = matmul_c(&sd, f, nao);
           // fd = F @ D
           let fd = matmul_c(f, d, nao);
           // fds = fd @ S
           let fds = matmul_c(&fd, s, nao);
           // err = sdf - fds (element-wise)
           sdf.iter().zip(&fds).map(|(a, b)| a - b).collect()
       }

       /// Square nao×nao C-order matrix multiply via pyscf-algebra::gemm.
       /// Helper used only inside err_vec_scf. Lives here (not in pyscf-algebra) because
       /// it has a fixed C-order convention specific to SCF Fock/Density layout.
       fn matmul_c(a: &[f64], b: &[f64], nao: usize) -> Vec<f64> {
           let mut out = vec![0.0_f64; nao * nao];
           // gemm(α=1, A, B, β=0, C) — caller is C-order so layout='C' (Phase 2 Pitfall 8).
           pyscf_algebra::gemm(1.0, a, b, 0.0, &mut out, nao, nao, nao)
               .expect("gemm on nao×nao square should not fail");
           out
       }
       ```

       ```rust
       use pyscf_diis::{Diis, DiisStorable};

       #[derive(Clone)]
       struct FlatVec(Vec<f64>);
       impl DiisStorable for FlatVec {
           fn as_flat(&self) -> &[f64] { &self.0 }
           fn from_flat(&mut self, s: &[f64]) { self.0.copy_from_slice(s); }

       }

       fn run_extrap() -> Vec<f64> {
           let mut diis = Diis::<FlatVec>::new(8);
           // Use a 256-element iterate so reduction has real parallelism opportunity.
           let f1 = FlatVec((0..256).map(|i| (i as f64).sin()).collect());
           let f2 = FlatVec((0..256).map(|i| (i as f64 + 1.0).sin()).collect());
           let f3 = FlatVec((0..256).map(|i| (i as f64 + 2.0).sin()).collect());
           let e1: Vec<f64> = (0..256).map(|i| (i as f64).cos()).collect();
           let e2: Vec<f64> = (0..256).map(|i| (i as f64 + 1.0).cos()).collect();
       ```

       ```rust
           hooks: &H,
           cfg: KernelConfig,
       ) -> Result<ScfResult, PyscfRsError> {
           tracing::info!(target: "pyscf_scf::kernel",
                          mol_nao = mol.nao_nr(),
       ```
