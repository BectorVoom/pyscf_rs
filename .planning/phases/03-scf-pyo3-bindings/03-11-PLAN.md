       //! Fock build + h_core + overlap (1e + 2e integrals via pyscf-gto::intor).
       use pyscf_core::{Density, Mole, PyscfRsError};

       pub fn default_get_hcore(mol: &Mole) -> Result<Density, PyscfRsError> {
           let t = pyscf_gto::intor(mol, "int1e_kin")?;
           let v = pyscf_gto::intor(mol, "int1e_nuc")?;
           let mut data = t.values.clone();
           pyscf_algebra::axpy(1.0, &v.values, &mut data)?;
           Ok(Density { nao: mol.nao_nr, data })
       }

       pub fn default_get_ovlp(mol: &Mole) -> Result<Density, PyscfRsError> {
           let s = pyscf_gto::intor(mol, "int1e_ovlp_sph")?;
           Ok(Density { nao: mol.nao_nr, data: s.values })
       }

       pub fn default_get_jk(mol: &Mole, dm: &Density) -> Result<(Density, Density), PyscfRsError> {
           // J[μν] = Σ_λσ (μν|λσ) D[λσ]
           // K[μν] = Σ_λσ (μλ|νσ) D[λσ]
           // Phase 2 plan 02-09 verification rollup is required for int2e_sph arity-4
           // dispatch. If still NotYetImplemented at land time, this plan's executor
           // adds a gap-closure note and routes through pyscf-df for the test corpus.
           let nao = mol.nao_nr;
           let eri = pyscf_gto::intor(mol, "int2e_sph")?;
           // Contract via pyscf-algebra::oracle_einsum for bit-exact J/K (Pitfall 9).
           // Executor: port pyscf/scf/jk.py:get_jk verbatim. The exact pattern is the
           // s8-symmetric eri layout; oracle_einsum expects flat vectors + a contraction
           // path string.
           let j = pyscf_algebra::oracle_einsum("μνλσ,λσ->μν", &[&eri.values, &dm.data],
                                                 &[nao, nao, nao, nao])?;
           let k = pyscf_algebra::oracle_einsum("μλνσ,λσ->μν", &[&eri.values, &dm.data],
                                                 &[nao, nao, nao, nao])?;
           Ok((Density { nao, data: j }, Density { nao, data: k }))
       }

       pub fn default_get_veff(mol: &Mole, dm: &Density) -> Result<Density, PyscfRsError> {
           // V_HF = J - 0.5 * K (RHF closed-shell)
           let (j, k) = default_get_jk(mol, dm)?;
           let mut veff = j.data.clone();
           pyscf_algebra::axpy(-0.5, &k.data, &mut veff)?;
           Ok(Density { nao: j.nao, data: veff })
       }

       pub fn default_get_fock(
           h1e: &Density, _s1e: &Density, vhf: &Density, _dm: &Density,
           cycle: i32, _diis_state: Option<&Density>,
       ) -> Result<Density, PyscfRsError> {

           // call into the pyscf-diis crate via a private adapter trait that this
           // function's caller (kernel_impl::scf_loop) is aware of.
           let mut data = h1e.data.clone();
           pyscf_algebra::axpy(1.0, &vhf.data, &mut data)?;
           if cycle >= 1 {
               tracing::warn!(
                   target: "pyscf_scf::fock",
                   "DIIS adapter slot — plan 03-04 not yet shipped, fixed-point fallback"
               );
           }
           Ok(Density { nao: h1e.nao, data })
       }

       use pyscf_core::{Density, MOCoefficients, PyscfRsError};

       pub fn default_make_rdm1(mo: &MOCoefficients) -> Result<Density, PyscfRsError> {
           let nao = mo.nao;
           let mut data = vec![0.0; nao * nao];
           for mu in 0..nao {

       //! SCF cycle loop — pyscf/scf/hf.py:48-244 verbatim port.
       use pyscf_core::{Density, Energy, Mole, PyscfRsError};
       use crate::{OverrideHooks, KernelConfig, ScfResult, InitGuessMode};
       use crate::error::ScfError;

       pub(crate) fn scf_loop<H: OverrideHooks>(
           mol: &Mole,
           hooks: &H,
           cfg: KernelConfig,
       ) -> Result<ScfResult, PyscfRsError> {
           tracing::info!(target: "pyscf_scf::kernel",
                          "SCF kernel entry (mol.nao={}, max_cycle={})",
                          mol.nao_nr, cfg.max_cycle);

           let s1e  = hooks.get_ovlp(mol)?;
           let h1e  = hooks.get_hcore(mol)?;
           let mut dm = match &cfg.init_guess {
               InitGuessMode::UserDM(d) => d.clone(),
               other => hooks.get_init_guess(mol, other)?,
           };
           let mut vhf = hooks.get_veff(mol, &dm)?;
           let (e_elec, _) = hooks.energy_elec(&dm, &h1e, &vhf)?;
           let e_nuc = mol.energy_nuc();
           let mut e_tot = Energy(e_elec.0 + e_nuc);
           let _conv_tol_grad = cfg.conv_tol_grad.unwrap_or_else(|| cfg.conv_tol.sqrt());

           let mut converged = false;
           let mut cycles = 0u32;
           let mut last_diff = f64::INFINITY;
           let mut mo_final: Option<pyscf_core::MOCoefficients> = None;

           for cycle in 0..cfg.max_cycle {
               cycles = cycle + 1;
               let fock = hooks.get_fock(&h1e, &s1e, &vhf, &dm, cycle as i32, None)?;
               let mut mo = hooks.eig(&fock, &s1e)?;
               let occ = hooks.get_occ(&mo.energies, mol.nelectron as usize)?;
               mo.occupations = occ;
               dm = hooks.make_rdm1(&mo)?;
               vhf = hooks.get_veff(mol, &dm)?;
               let (e_elec, _) = hooks.energy_elec(&dm, &h1e, &vhf)?;
               let e_new = Energy(e_elec.0 + e_nuc);
               last_diff = (e_new.0 - e_tot.0).abs();
               tracing::debug!(target: "pyscf_scf::kernel",
                               cycle = cycle, e_tot = e_new.0, delta = last_diff);
               e_tot = e_new;
               mo_final = Some(mo);
               if last_diff < cfg.conv_tol {
                   converged = true;
                   break;
               }
           }
           let mo = mo_final.ok_or_else(|| ScfError::ConvergenceFailure {
               cycles: 0, last_diff: f64::INFINITY
           })?;
           if !converged {
               return Err(ScfError::ConvergenceFailure { cycles, last_diff }.into());
           }

       pub fn mulliken_pop(rhf: &RHF) -> Result<MullikenResult, PyscfRsError> {
           // Algorithm (hf.py:1262-1300):
           //   1. Get DM and S
           //   2. AO populations: pop[μ] = (D · S)[μ, μ]
           //   3. Atom charges: chg[A] = Z_A - Σ_{μ ∈ atom A} pop[μ]

               pyscf_core::CoreError::Other("mulliken_pop: kernel not run yet".into())
           ))?;

           for mu in 0..nao {
               // (D · S)[μ, μ] = Σ_ν D[μ, ν] · S[ν, μ]
               ao_pop[mu] = pyscf_algebra::oracle_sum(
                   (0..nao).map(|nu| dm.data[mu * nao + nu] * s.data[nu * nao + mu])
               );
           }
           // Aggregate AO populations onto atoms via mol's ao_loc_nr.
           let natm = rhf.mol.natm;
           let mut atom_charges = vec![0.0; natm];
           // For each atom A: chg[A] = Z_A - Σ_{μ ∈ atom A} ao_pop[μ]
           // Executor: use `rhf.mol.ao_labels()` or `mol.aoslice_by_atom()` (Phase 2 shipped).

               let pop_a: f64 = pyscf_algebra::oracle_sum((lo..hi).map(|mu| ao_pop[mu]));
               let z = rhf.mol.atom_charge(a) as f64;
               atom_charges[a] = z - pop_a;
           }

           // dip = -Tr(D · int1e_r) + Σ_atoms Z * r_atom
           let mo = rhf.mo_coeff.as_ref().ok_or_else(|| PyscfRsError::Core(

           let mut dip = [0.0_f64; 3];
