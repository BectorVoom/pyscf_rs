"""Phase 3 SCF-01 — RHF on H2O/cc-pVDZ ≤ 1 µHartree vs upstream."""
import sys
import importlib.util
import numpy as np
from pyscf import gto, scf  # this is pyscf-rs's overlay (BIND-02)

# Load upstream pyscf (the upstream `pyscf/` tree) under a different namespace.
# In a maturin-develop install, the upstream tree still lives at <repo>/pyscf/;
# we can import it via the file path.
def upstream_pyscf():
    # If repo root is the cwd, upstream is at ./pyscf/. Otherwise resolve via __file__.
    upstream_init = "<repo>/pyscf/__init__.py"  # executor: detect via os.path
    spec = importlib.util.spec_from_file_location("_upstream_pyscf", upstream_init)
    mod = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(mod)
    return mod

def test_rhf_h2o_ccpvdz_u_hartree():
    mol_rs   = gto.M(atom='O 0 0 0; H 0.757 0.587 0; H -0.757 0.587 0', basis='cc-pvdz')
    mf_rs    = scf.RHF(mol_rs).run()
    assert mf_rs.converged

    up = upstream_pyscf()
    mol_up = up.gto.M(atom='O 0 0 0; H 0.757 0.587 0; H -0.757 0.587 0', basis='cc-pvdz')
    mf_up  = up.scf.RHF(mol_up).run()
    diff = abs(mf_rs.e_tot - mf_up.e_tot)
    assert diff < 1e-6, f"|e_rs - e_up| = {diff:.3e} > 1 µHartree"
```

       """Phase 3 BIND-04 — NumPy zero-copy boundary discipline (Pitfall 5).

       Source: RESEARCH §"Pattern 4 stride-fuzz test" lines 608-635 (verbatim).
       Tests that mf.get_veff accepts 4 stride-variant views of the same density
       matrix and returns bit-identical output.
       """
       import numpy as np
       from pyscf import scf


       def test_get_veff_accepts_4_stride_variants(h2o_mol):
           mol = h2o_mol
           mf = scf.RHF(mol).run()
           nao = mol.nao_nr()
           dm = np.asarray(mf.make_rdm1())  # canonical C-contiguous

           # Build 4 stride-equivalent views of the SAME data:
           view_a = dm                                                       # C-contig
           view_b = dm.T                                                     # F-contig (transposed)
           # view_c: strided via odd-step slicing (force non-contiguous)
           bigger = np.zeros((nao * 2, nao * 2))
           bigger[::2, ::2] = dm
           view_c = bigger[::2, ::2]
           # view_d: offset C-strided view
           wider = np.zeros((nao, nao + 5))
           wider[:, 1 : nao + 1] = dm
           view_d = wider[:, 1 : nao + 1]

           # All 4 must produce the same get_veff (BIND-04 bit-identical contract).
           # NOTE: dm.T contains the TRANSPOSE of dm, not the same data. The contract is:
           # "the boundary converter copies into a canonical C-order Density before
           # invoking the kernel" — so dm.T would produce get_veff(D.T), NOT get_veff(D).
           # For the symmetric case (D == D.T, which is true for valid density matrices),
           # the result is the same. The stride-fuzz contract holds for valid symmetric
           # density input.
           assert np.allclose(dm, dm.T), "test fixture: density must be symmetric"
           out_a = np.asarray(mf.get_veff(mol, view_a))
           out_b = np.asarray(mf.get_veff(mol, view_b))
           out_c = np.asarray(mf.get_veff(mol, view_c))
           out_d = np.asarray(mf.get_veff(mol, view_d))

           # Bit-identical: NumPy testing assert_array_equal (NOT allclose).
           np.testing.assert_array_equal(out_a, out_b, "view_b (dm.T) gives different result")
           np.testing.assert_array_equal(out_a, out_c, "view_c ([::2]) gives different result")
           np.testing.assert_array_equal(out_a, out_d, "view_d ([:, 1:nao+1]) gives different result")
       ```
