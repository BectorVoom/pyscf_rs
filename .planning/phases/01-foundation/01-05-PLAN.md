Purpose: These five lints are the "forcing function for code review" (RESEARCH §2). They catch:
- **check-no-fma** (FOUND-05; Pitfall 1 SHOWSTOPPER): FMA contraction in oracle-profile machine code; compromises bit-exact reproducibility against PySCF.
- **check-forbidden-paths** (FOUND-08; Pitfall 21): scope-creep imports from upstream PySCF modules out of scope (pbc/x2c/mcscf/tdscf/adc/gw/eom/nac/eph).
- **check-catch-unwind** (FOUND-07; Pitfall 14): unsafe FFI surface where Rust panics could escape across the C boundary causing UB.
- **check-dependency-wall** (ALG-06): method crates (gto/scf/dft/...) accidentally pulling cubecl-* deps; would defeat the architectural separation.
- **check-cubecl-pin** (FOUND-04; Pitfall 3): cubecl 0.10.0 lockstep with cintx/libxc_rs/xcfun_rs; drift causes ABI breaks.

Output: `xtask/` directory with 6 files (Cargo.toml + main.rs + 5 binaries). Each binary is a standalone, deterministic, sub-second (after compile) lint.

    ```rust
    //!   cargo run -p xtask                # runs all 5 checks
    //!   cargo run -p xtask --bin check-no-fma   # one specific check

    fn main() -> ExitCode {

            "check-forbidden-paths",
            "check-catch-unwind",
            "check-dependency-wall",
            "check-cubecl-pin",

        for check in &checks {

            let status = Command::new("cargo")
                .args(["run", "--quiet", "-p", "xtask", "--bin", check])
                .status();

                eprintln!("--- xtask: {check} FAILED ---");
                all_passed = false;
            } else {
                eprintln!("--- xtask: {check} OK ---");
            }
        }
        if all_passed { ExitCode::SUCCESS } else { ExitCode::from(2) }
    }
    ```

    ```rust
    //! FOUND-04: cubecl 0.10.0 lockstep verification across siblings.
    //! Adapted from ~/Documents/workspace/xcfun_rs/xtask/src/bin/check_cubecl_pin.rs
    //! (PATTERNS row "check_cubecl_pin"). Adds cubecl-matmul/cubecl-reduce
    //! at the 0.9.0-pre.5 ABI per RESEARCH §"Standard Stack" Pitfall 1.

    use anyhow::{Context, Result};
    use serde_json::Value;
    use std::path::PathBuf;
    use std::process::{Command, ExitCode};

    /// cubecl-{cpu,cuda,hip,wgpu,runtime} all pinned at this version.
    const REQUIRED_VERSION: &str = "0.10.0";

    /// Crates that MUST equal `REQUIRED_VERSION`.
    const PINNED_CRATES: &[&str] = &[
        "cubecl",
        "cubecl-cpu",
        "cubecl-cuda",
        "cubecl-hip",
        "cubecl-runtime",
        "cubecl-wgpu",
    ];

    /// cubecl-matmul / cubecl-reduce 0.10.0 unpublished; pinned at the
    /// pre-release that has the matching ABI (RESEARCH §"Standard Stack").
    const PRE_REQUIRED_VERSION: &str = "0.9.0-pre.5";
    const PRE_PINNED_CRATES: &[&str] = &[
        "cubecl-matmul",
        "cubecl-reduce",
    ];

    fn main() -> Result<ExitCode> {
        let root = workspace_root()?;
        let output = Command::new("cargo")
            .current_dir(&root)
            .args(["metadata", "--format-version", "1", "--no-deps"])
            .output()
            .context("failed to spawn `cargo metadata`")?;
        if !output.status.success() {
            anyhow::bail!("cargo metadata failed: {}",
                String::from_utf8_lossy(&output.stderr));
        }
        let metadata: Value = serde_json::from_slice(&output.stdout)
            .context("parse cargo metadata JSON")?;
        let empty: Vec<Value> = Vec::new();
        let packages = metadata["packages"].as_array().unwrap_or(&empty);

        let mut violations = Vec::new();
        // Walk the resolved dep graph (not just workspace members) — cargo
        // metadata --no-deps only lists workspace members, so we need to
        // re-run without --no-deps to see the cubecl deps.
        let full = Command::new("cargo")
            .current_dir(&root)
            .args(["metadata", "--format-version", "1"])
            .output()
            .context("failed to spawn `cargo metadata` (full graph)")?;
        let full_meta: Value = serde_json::from_slice(&full.stdout)
            .context("parse full cargo metadata JSON")?;
        let full_pkgs = full_meta["packages"].as_array().unwrap_or(&empty);

        for pkg in full_pkgs {
            let name = pkg["name"].as_str().unwrap_or("");
            let version = pkg["version"].as_str().unwrap_or("");
            if PINNED_CRATES.contains(&name) && version != REQUIRED_VERSION {
                violations.push(format!("{name}: version {version} (expected {REQUIRED_VERSION})"));
            }
            if PRE_PINNED_CRATES.contains(&name) && version != PRE_REQUIRED_VERSION {
                violations.push(format!("{name}: version {version} (expected {PRE_REQUIRED_VERSION})"));
            }
        }

        let _ = packages;  // silence unused on no-deps fallback path
        if violations.is_empty() {
            eprintln!("check-cubecl-pin: PASS — all cubecl-* crates at expected version (FOUND-04)");
            Ok(ExitCode::SUCCESS)
        } else {
            eprintln!("check-cubecl-pin: FAIL — cubecl version drift detected (FOUND-04 / Pitfall 1):");
            for v in &violations { eprintln!("  - {v}"); }
            eprintln!("\nFix: align workspace [workspace.dependencies] cubecl-* pins, or update");
            eprintln!("the sibling-crate revs in [patch.crates-io] to ones that use the matching pin.");
            Ok(ExitCode::from(2))
        }
    }

    fn workspace_root() -> Result<PathBuf> {
        let mut p = std::env::current_dir().context("getcwd")?;
        loop {
            if p.join("Cargo.toml").is_file() && p.join("Cargo.toml").metadata().is_ok() {

            }
            if !p.pop() { anyhow::bail!("no workspace root found from {:?}", std::env::current_dir()?); }
        }
    }
    ```

    **(B) `xtask/src/bin/check_dependency_wall.rs`** — ALG-06 dep-wall, denylist semantics (inverse of xcfun_rs `check_boundaries.rs`):
    ```rust
    //! ALG-06: only pyscf-algebra and pyscf-runtime may declare cubecl-*
    //! deps. Adapted from xcfun_rs/xtask/src/bin/check_boundaries.rs
    //! (PATTERNS row "check_dependency_wall") — switches from allowlist
    //! to denylist semantics. Walks `cargo metadata --format-version 1
    //! --no-deps` and fails if any other crate declares a normal cubecl-*
    //! dep.
    //!
    //! Note: dev-dependencies are NOT checked (they don't affect the
    //! shipped wheel). RESEARCH Pitfall 4 documents this gap; future work
    //! may extend the lint.

    use anyhow::{Context, Result};
    use serde_json::Value;
    use std::path::PathBuf;
    use std::process::{Command, ExitCode};

    /// Crates that may NOT appear as a normal dep of any pyscf-rs crate
    /// other than the ALG-06 carve-out crates.
    const FORBIDDEN_DEPS: &[&str] = &[
        "cubecl",
        "cubecl-cpu",
        "cubecl-cuda",
        "cubecl-hip",
        "cubecl-matmul",
        "cubecl-reduce",
        "cubecl-runtime",
        "cubecl-std",
        "cubecl-wgpu",
    ];

    /// Crates permitted to consume cubecl-* (ALG-06 carve-out).
    const ALLOWED_CRATES: &[&str] = &[
        "pyscf-algebra",
        "pyscf-runtime",
    ];

    fn main() -> Result<ExitCode> {
        let root = workspace_root()?;
        let output = Command::new("cargo")
            .current_dir(&root)
            .args(["metadata", "--format-version", "1", "--no-deps"])
            .output()
            .context("failed to spawn `cargo metadata --no-deps`")?;
        if !output.status.success() {
            anyhow::bail!("cargo metadata failed: {}", String::from_utf8_lossy(&output.stderr));
        }
        let metadata: Value = serde_json::from_slice(&output.stdout)
            .context("parse cargo metadata JSON")?;
        let empty: Vec<Value> = Vec::new();
        let packages = metadata["packages"].as_array().unwrap_or(&empty);

        let mut violations = Vec::new();
        for pkg in packages {
            let name = pkg["name"].as_str().unwrap_or("");
            // Skip the carve-out crates and non-pyscf packages.
            if ALLOWED_CRATES.contains(&name) { continue; }
            // Only check pyscf-rs workspace crates (skip xtask, the
            // top-level pyscf-rs façade is allowed since it transitively
            // re-exports algebra; the dep-wall concern is only that
            // method crates don't directly name cubecl-*).
            if !name.starts_with("pyscf-") { continue; }
            // pyscf-rs façade re-exports pyscf-algebra (path dep), but its
            // OWN [dependencies] block is allowed to include pyscf-algebra
            // — the wall is about cubecl-*, not pyscf-algebra. So no
            // exception needed here.

            let deps = pkg["dependencies"].as_array().unwrap_or(&empty);
            for dep in deps {
                let dep_name = dep["name"].as_str().unwrap_or("");
                let kind = dep["kind"].as_str();    // null = normal, "dev"|"build" = others
                let is_normal = kind.is_none() || kind == Some("");
                if !is_normal { continue; }
                if FORBIDDEN_DEPS.contains(&dep_name) {
                    violations.push(format!(
                        "{name}: declares normal dep on `{dep_name}` — ALG-06 forbids; only {ALLOWED_CRATES:?} may consume cubecl-*"
                    ));
                }
            }
        }

        if violations.is_empty() {
            eprintln!("check-dependency-wall: PASS — cubecl-* containment intact (ALG-06)");
            Ok(ExitCode::SUCCESS)
        } else {
            eprintln!("check-dependency-wall: FAIL — ALG-06 violation:");
            for v in &violations { eprintln!("  - {v}"); }
            eprintln!("\nFix: route through pyscf-algebra's public surface (Tensor + free fns).");
            eprintln!("Reference: docs/manual/Cubecl/ + RESEARCH.md Architecture Patterns.");
            Ok(ExitCode::from(2))
        }
    }

    fn workspace_root() -> Result<PathBuf> {
        let mut p = std::env::current_dir().context("getcwd")?;
        loop {
            if p.join("Cargo.toml").is_file() {
                let s = std::fs::read_to_string(p.join("Cargo.toml"))?;
                if s.contains("[workspace]") { return Ok(p); }
            }
            if !p.pop() { anyhow::bail!("no workspace root found"); }
        }
    }
    ```
