def make_atm_env(atom, ptr=0, nucmod=NUC_POINT, nucprop=None):
    """Convert atom to libcint internal _atm row + _env append."""
    nuc_charge = _charge(atom[0])
    _env = numpy.hstack(([0]*ptr, atom[1], [nucmod_to_zeta(nucmod, nuc_charge, nucprop)]))
    _atm = numpy.zeros(ATM_SLOTS, dtype=numpy.int32)
    _atm[CHARGE_OF]  = nuc_charge
    _atm[PTR_COORD]  = ptr
    _atm[NUC_MOD_OF] = nucmod
    _atm[PTR_ZETA]   = ptr + 3
    return _atm, _env
```

def make_bas_env(basis, atom_id=0, ptr=0):
    """Convert per-element basis to _bas rows + _env append (exponents + coeffs)."""
    _bas = []
    _env = []
    for shell in basis:                         # shell = (l, *primitives)
        l = shell[0]
        primitives = numpy.array(shell[1:])     # shape (nprim, 1+nctr)
        nprim, nctr = primitives.shape[0], primitives.shape[1] - 1
        exponents = primitives[:, 0]
        coeffs = primitives[:, 1:]
        # NORMALISE: per-primitive radial gto_norm + per-contraction _nomalize_contracted_ao
        norm = gto_norm(l, exponents)            # shape (nprim,)
        coeffs = coeffs * norm[:, None]          # broadcast normalisation
        coeffs = _nomalize_contracted_ao(l, exponents, coeffs)
        # Append to _env in F-order: exps then coeffs (column-major)
        _env.extend(exponents)
        _env.extend(coeffs.T.flatten())          # F-order coeff matrix
        # Append _bas row
        bas_row = [atom_id, l, nprim, nctr, 0,    # KAPPA_OF=0 for sph/cart
                   ptr, ptr + nprim, 0]
        _bas.append(bas_row)
        ptr += nprim + nprim * nctr
    return _bas, _env, ptr
```

```python
def make_env(atoms, basis, pre_env=[], nucmod=None, nucprop=None):
    _env = list(pre_env) + [0]*PTR_ENV_START   # libcint reserved slots
    _atm = []
    _bas = []
    # === Pass 1: per-atom _atm + xyz to _env ===
    for atom_id, atom in enumerate(atoms):
        nm = nucmod.get(atom[0], NUC_POINT) if nucmod else NUC_POINT
        nv = nucprop.get(atom[0], None) if nucprop else None
        atm_row, env_chunk = make_atm_env(atom, len(_env), nm, nv)
        _atm.append(atm_row)
        _env.extend(env_chunk[len(_env)-len(env_chunk)+len(_env):])   # tail of env_chunk
    # === Per-symbol basis template (Pitfall 4: first-occurrence order) ===
    basdic = {}                                  # symbol → bas_template_rows
    for atom_id, atom in enumerate(atoms):
        symb = atom[0]
        if symb not in basdic:
            bas_rows, env_chunk, _ = make_bas_env(basis[symb], atom_id=0, ptr=len(_env))
            basdic[symb] = bas_rows
            _env.extend(env_chunk)
    # === Pass 2: per-atom clone the per-symbol template, patch ATOM_OF ===
    for atom_id, atom in enumerate(atoms):
        symb = atom[0]
        for bas_template in basdic[symb]:
            bas_row = list(bas_template)
            bas_row[ATOM_OF] = atom_id
            _bas.append(bas_row)
    return numpy.array(_atm), numpy.array(_bas), numpy.array(_env)
```

    //! GTO-04: make_env — flat-array projection per D-03.
    //!
    //! Source: pyscf/gto/mole.py:961-1105 (Apache-2.0).
    //! Slot constants from cintx_compat::raw via pyscf_core::raw_layout per D-03.

    use pyscf_core::raw_layout::*;
    use pyscf_core::{Mole, ParsedAtom, ParsedBasis, ShellSpec};
    use std::collections::HashMap;

    /// Output of make_env: the four flat arrays + nao_nr.
    #[derive(Debug, Default)]
    pub struct MakeEnvOutput {
        pub _atm: Vec<i32>,
        pub _bas: Vec<i32>,
        pub _env: Vec<f64>,
        pub ao_loc_nr: Vec<i32>,
        pub nao_nr: usize,
    }

    /// Project (atoms, basis_per_symbol) to libcint flat arrays.
    /// Runs the upstream two-pass algorithm:
    ///   Pass 1: per-atom _atm + xyz to _env
    ///   Per-symbol template (Pitfall 4 first-occurrence order)
    ///   Pass 2: clone per-symbol bas template per atom, patch ATOM_OF
    pub fn make_env(
        atoms: &[ParsedAtom],
        basis: &HashMap<String, ParsedBasis>,
        cart: bool,
    ) -> MakeEnvOutput {
        let mut out = MakeEnvOutput::default();
        // Reserve the libcint global-param slots (PTR_ENV_START = 20 zeros).
        out._env = vec![0.0_f64; PTR_ENV_START];

        // ===== Pass 1: per-atom _atm rows + xyz to _env =====
        for (sym_with_suffix, xyz) in atoms {
            let alpha: String = sym_with_suffix.chars().take_while(|c| c.is_alphabetic()).collect();
            let charge = crate::format_atom::charge_for_symbol(&alpha).unwrap_or(0);
            let nuclear_model = POINT_NUC;     // v1: always point nucleus (NUC_MOD_OF=1)
            let zeta = 0.0_f64;                // unused for point nuclei

            let ptr_coord = out._env.len();
            // Append xyz (3 doubles)
            out._env.extend_from_slice(xyz);
            // Append zeta (1 double; placeholder for finite-nucleus model)
            out._env.push(zeta);

            // Append _atm row (6 i32)
            let row_start = out._atm.len();
            out._atm.resize(row_start + ATM_SLOTS, 0);
            out._atm[row_start + CHARGE_OF]  = charge;
            out._atm[row_start + PTR_COORD]  = ptr_coord as i32;
            out._atm[row_start + NUC_MOD_OF] = nuclear_model;
            out._atm[row_start + PTR_ZETA]   = (ptr_coord + 3) as i32;
            // Slots 4 (PTR_FRAC_CHARGE), 5 stay zero per upstream.
        }

        // ===== Per-symbol bas template (Pitfall 4: first-occurrence order) =====
        // basdic: symbol → Vec<bas_row_template (8 i32 each)>
        let mut basdic: HashMap<String, Vec<[i32; BAS_SLOTS]>> = HashMap::new();
        let mut symbol_order: Vec<String> = Vec::new();
        for (sym_with_suffix, _) in atoms {
            let alpha: String = sym_with_suffix.chars().take_while(|c| c.is_alphabetic()).collect();
            let upper = alpha.to_ascii_uppercase();
            if basdic.contains_key(&upper) { continue; }
            symbol_order.push(upper.clone());
            // Build bas template + append exponents/coeffs to _env for THIS symbol.
            let parsed = basis.get(&upper).expect("format_basis ensures every atom symbol is in basis map");
            let templates = make_bas_env_for_symbol(parsed, &mut out._env);
            basdic.insert(upper, templates);
        }

        // ===== Pass 2: per-atom clone the per-symbol template, patch ATOM_OF =====
        for (atom_id, (sym_with_suffix, _)) in atoms.iter().enumerate() {
            let alpha: String = sym_with_suffix.chars().take_while(|c| c.is_alphabetic()).collect();
            let upper = alpha.to_ascii_uppercase();
            let templates = &basdic[&upper];
            for tpl in templates {
                let mut row = *tpl;
                row[ATOM_OF] = atom_id as i32;
                out._bas.extend_from_slice(&row);
            }
        }

        // Assert dimensional invariants (Pitfall 2)
        debug_assert_eq!(out._atm.len() % ATM_SLOTS, 0);
        debug_assert_eq!(out._bas.len() % BAS_SLOTS, 0);

        // Compute nao_nr + ao_loc_nr from _bas.
        let nbas = out._bas.len() / BAS_SLOTS;
        let mut ao_loc = Vec::with_capacity(nbas + 1);
        let mut acc: i32 = 0;
        ao_loc.push(0);
        for i in 0..nbas {
            let l = out._bas[i * BAS_SLOTS + ANG_OF] as u8;
            let nctr = out._bas[i * BAS_SLOTS + NCTR_OF] as i32;
            let dim_per_ctr = if cart {
                ((l as i32 + 1) * (l as i32 + 2)) / 2
            } else {
                2 * l as i32 + 1
            };
            acc += dim_per_ctr * nctr;
            ao_loc.push(acc);
        }
        out.nao_nr = acc as usize;
        out.ao_loc_nr = ao_loc;

        out
    }

    /// Per-symbol: build bas templates + append normalised exponents/coeffs to _env.
    /// Returns Vec<bas_row_template> with PTR_EXP / PTR_COEFF pointing into _env;
    /// ATOM_OF stays 0 (caller patches per atom).
    fn make_bas_env_for_symbol(
        parsed: &ParsedBasis,
        _env: &mut Vec<f64>,
    ) -> Vec<[i32; BAS_SLOTS]> {
        let mut templates = Vec::with_capacity(parsed.shells.len());
        for shell in &parsed.shells {
            let l = shell.l;
            let nprim = shell.exponents.len();
            let nctr  = shell.coeffs.len();   // each row in coeffs is one contraction column

            // Apply normalisation transforms: gto_norm + _nomalize_contracted_ao.
            // The result `final_coeffs` is shape (nctr, nprim) — F-order on flatten gives
            // column-major coefficient matrix consumed by libcint.
            let final_coeffs = normalise_contractions(l, &shell.exponents, &shell.coeffs);

            let ptr_exp = _env.len();
            _env.extend_from_slice(&shell.exponents);
            let ptr_coeff = _env.len();
            // Append in F-order: for each contraction column c, append nprim primitives.
            for c_idx in 0..nctr {
                _env.extend_from_slice(&final_coeffs[c_idx]);
            }

            let mut row = [0_i32; BAS_SLOTS];
            row[ATOM_OF]   = 0;                        // patched per atom in pass 2
            row[ANG_OF]    = l as i32;
            row[NPRIM_OF]  = nprim as i32;
            row[NCTR_OF]   = nctr as i32;
            row[KAPPA_OF]  = 0;                        // Pitfall 5: always 0 for sph/cart in v1
            row[PTR_EXP]   = ptr_exp as i32;
            row[PTR_COEFF] = ptr_coeff as i32;
            // slot 7 stays 0 per upstream

            templates.push(row);
        }
        templates
    }

    /// Per-primitive radial gto_norm + per-contraction _nomalize_contracted_ao.
    /// Source: pyscf/gto/mole.py:120-155 (gto_norm) + 1018-1027 (_nomalize_contracted_ao).
    ///
    /// Returns coefficients indexed [contraction_column][primitive] (F-order on flatten).
    fn normalise_contractions(l: u8, exponents: &[f64], raw_coeffs: &[Vec<f64>]) -> Vec<Vec<f64>> {
        let nprim = exponents.len();
        let nctr = raw_coeffs.len();

        // Per-primitive radial normalisation factor.
        let prim_norm: Vec<f64> = exponents.iter().map(|&a| gto_norm(l, a)).collect();

        // Apply per-primitive norm: c'_{ip} = c_{ip} * prim_norm[p]
        // raw_coeffs[c_idx][p_idx] = c_{c, p}; pyscf coeffs after gto_norm:
        let mut scaled: Vec<Vec<f64>> = (0..nctr)
            .map(|c| (0..nprim).map(|p| raw_coeffs[c][p] * prim_norm[p]).collect())
            .collect();

        // _nomalize_contracted_ao: post-multiply each contraction column c by 1/sqrt(c.T @ S @ c)
        // where S[i,j] = gaussian_int(2l+2, e_i + e_j).
        for c in 0..nctr {
            let col = &scaled[c];
            let mut norm_sq = 0.0_f64;
            for i in 0..nprim {
                for j in 0..nprim {
                    norm_sq += col[i] * col[j] * gaussian_int(2 * l as i32 + 2, exponents[i] + exponents[j]);
                }
            }
            if norm_sq > 0.0 {
                let inv = 1.0 / norm_sq.sqrt();
                for p in 0..nprim {
                    scaled[c][p] *= inv;
                }
            }
        }
        scaled
    }

    /// Gaussian integral `int_0^inf r^n exp(-α r²) dr = 0.5 * Γ((n+1)/2) / α^((n+1)/2)`.
    /// Source: pyscf/gto/mole.py `gaussian_int` helper (~mole.py:120).
    /// Phase 2 ships an exact closed form; pulls f64::gamma equivalent via libm.
    fn gaussian_int(n: i32, alpha: f64) -> f64 {
        let half_n_plus_1 = (n as f64 + 1.0) * 0.5;
        // Γ((n+1)/2): for integer (n+1)/2 → factorial; for half-integer → sqrt(π) * (2k-1)!! / 2^k
        // Use libm::tgamma via std (Rust std exposes f64::gamma via crate libm or unstable).
        // Stable workaround: Stirling-based or recursive — but for normalisation factors
        // n is small (≤ ~20 for l≤6). Use the recursive Γ(x) = (x-1) * Γ(x-1):
        let gamma = tgamma(half_n_plus_1);
        0.5 * gamma / alpha.powf(half_n_plus_1)
    }

    /// Per-primitive radial normalisation:
    /// `gto_norm(l, α) = 1 / sqrt(gaussian_int(2l+2, 2α))`.
    /// Source: pyscf/gto/mole.py:125-155.
    fn gto_norm(l: u8, alpha: f64) -> f64 {
        1.0 / gaussian_int(2 * l as i32 + 2, 2.0 * alpha).sqrt()
    }

    /// Γ(x) — Lanczos approximation, sufficient accuracy for half-integer args used here.
    /// (For Phase 2 normalisation only; if precision becomes a concern, swap to libm::tgamma.)
    fn tgamma(x: f64) -> f64 {
        // libm::tgamma is the right call here.
        // Add to Cargo.toml if not already: libm.workspace = true
        libm::tgamma(x)
    }

    //! access by 02-05 intor + 02-06 eval_gto + downstream phase consumers.

    pub fn build_cintx_basis_set(
        atoms: &[ParsedAtom],
        basis: &HashMap<String, ParsedBasis>,
    ) -> Result<Arc<BasisSet>, PyscfRsError> {
        // Build cintx Atoms.
        let cintx_atoms: Vec<Atom> = atoms.iter().map(|(sym, xyz)| {
            let alpha: String = sym.chars().take_while(|c| c.is_alphabetic()).collect();
            let charge = crate::format_atom::charge_for_symbol(&alpha).unwrap_or(0) as u8;
            Atom {
                symbol: sym.clone(),
                charge,
                coord: *xyz,
                nuclear_model: CintxNuc::Point,   // v1 always point nuclei
            }
        }).collect();

        // Build cintx Shells in per-symbol-first-occurrence order, cloned per atom.
        let mut cintx_shells: Vec<Arc<Shell>> = Vec::new();
        let mut symbol_first_seen: HashMap<String, ()> = HashMap::new();
        let mut symbol_order: Vec<String> = Vec::new();
        for (sym, _) in atoms {
            let alpha: String = sym.chars().take_while(|c| c.is_alphabetic()).collect();
            let upper = alpha.to_ascii_uppercase();
            if symbol_first_seen.insert(upper.clone(), ()).is_none() {
                symbol_order.push(upper);
            }
        }
        for (atom_id, (sym, _)) in atoms.iter().enumerate() {
            let alpha: String = sym.chars().take_while(|c| c.is_alphabetic()).collect();
            let upper = alpha.to_ascii_uppercase();
            let parsed = basis.get(&upper).expect("format_basis populated map");
            for spec in &parsed.shells {
                // Flatten coefficients to F-order (prim outer, ctr inner) — matches cintx Shell layout.
                // Verify by reading cintx/crates/cintx-core/src/shell.rs the exact field order.
                let coeffs_flat: Vec<f64> = (0..spec.coeffs.len())
                    .flat_map(|c_idx| spec.exponents.iter().enumerate()
                        .map(move |(p_idx, _)| spec.coeffs[c_idx][p_idx]))
                    .collect();
                cintx_shells.push(Arc::new(Shell {
                    atom_index: atom_id,
                    angular_momentum: spec.l,
                    exponents: spec.exponents.clone(),
                    coefficients: coeffs_flat,
                    n_contractions: spec.coeffs.len(),
                    n_primitives: spec.exponents.len(),
                    kappa: 0,                          // Pitfall 5: always 0 for sph/cart
                }));
            }
        }

        let meta = Default::default();   // BasisMeta default — verify cintx_core::basis::BasisMeta has impl Default
        Ok(Arc::new(BasisSet::new(cintx_atoms, cintx_shells, meta)))
    }
