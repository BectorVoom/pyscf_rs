        ```rust
        //! GTO-05 (eval half): EcpEngine impl backed by cintx ECP integral evaluation.
        //! Replaces EcpEngineNotAvailable per gap-closure plan 02-10.

        use cintx_core::{OperatorId, Representation, ShellTuple};
        use cintx_rs::{ExecutionOptions, SessionRequest};
        use pyscf_core::{CoreError, Density, EcpEngine, Mole, PyscfRsError};

        /// EcpEngine backed by cintx ECP. Routes int1e_ecp* names through cintx_rs.
        #[derive(Debug, Default, Clone, Copy)]
        pub struct CintxEcpEngine;

        impl EcpEngine for CintxEcpEngine {
            fn ecp_int1e(&self, mol: &Mole, name: &str) -> Result<Density, PyscfRsError> {
                let basis_arc = mol.cintx_basis()?;
                let basis_ref = &*basis_arc;
                let nbas = mol.nbas;
                let shells = ShellTuple::from_indices(basis_ref, &(0..nbas).collect::<Vec<_>>())
                    .map_err(|e| PyscfRsError::Core(CoreError::InvalidMolecule(
                        format!("ShellTuple for ECP failed: {:?}", e)
                    )))?;
                // Read cintx_core::operator post-merge for the exact OperatorId construction
                // for ECP-family names. Likely: OperatorId::for_symbol(name) covers it via
                // the same dispatch the non-ECP path uses.
                let operator = OperatorId::for_symbol(name);
                let representation = if name.ends_with("_cart") {
                    Representation::Cart
                } else {
                    Representation::Spheric
                };
                let request = SessionRequest::new(
                    operator, representation, basis_ref, shells, ExecutionOptions::default(),
                );
                let outcome = request.query_workspace()
                    .map_err(|e| PyscfRsError::Core(CoreError::InvalidMolecule(
                        format!("cintx ECP workspace query failed: {:?}", e)
                    )))?
                    .evaluate()
                    .map_err(|e| PyscfRsError::Core(CoreError::InvalidMolecule(
                        format!("cintx ECP evaluate failed: {:?}", e)
                    )))?;
                // Convert IntegralTensor to Density. The exact API depends on
                // pyscf_core::Density's constructors — read crates/pyscf-core/src/density.rs.
                Ok(Density::from_flat(outcome.tensor.owned_values, outcome.tensor.extents))
            }
        }
        ```
        NOTE: If `Density::from_flat` doesn't exist, add it to `pyscf-core::density`
        in this plan as a small helper.
