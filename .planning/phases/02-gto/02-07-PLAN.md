```python
def make_ecp_env(mol, _atm, ecp, pre_env=[]):
    """Append ECP shells to _ecpbas + ECP exponents/coefficients to _env.
    Also CHARGE_OF subtracts n_core for ECP atoms.
    Layout: _ecpbas rows are 8 i32 each (same BAS_SLOTS=8 as _bas).
    """
    _env = list(pre_env)
    _ecpbas = []
    for symb, ecp_data in ecp.items():
        n_core, channels = ecp_data
        for atom_id, atom_symb in enumerate([a[0] for a in mol._atom]):
            if _atom_strip(atom_symb) != _atom_strip(symb): continue
            # Subtract core electrons from CHARGE_OF
            _atm[atom_id, CHARGE_OF] -= n_core
            for ch in channels:
                # ch.l, ch.n_powers, ch.exponents, ch.coeffs
                ptr_exp   = len(_env)
                _env.extend(ch.exponents)
                ptr_coeff = len(_env)
                _env.extend(ch.coeffs)
                _ecpbas.append([atom_id, ch.l, len(ch.exponents), 1, 0, ptr_exp, ptr_coeff, 0])
    return numpy.array(_atm), numpy.array(_ecpbas), numpy.array(_env)
```

    //! GTO-05 (loading half): format_ecp + make_ecp_env.
    //!
    //! Source: pyscf/gto/mole.py:1107-1160 + pyscf/gto/basis/__init__.py:730-779.
    //! D-06: loading ships in Phase 2; evaluation closes via gap-closure plan 02-10.

    use crate::basis;
    use crate::types::EcpInput;
    use pyscf_core::raw_layout::*;
    use pyscf_core::{BasisLoadError, EcpLoadError, EcpShell, ParsedAtom, ParsedEcp, PyscfRsError};
    use std::collections::HashMap;

    /// Resolve EcpInput → per-element-symbol ParsedEcp map. Mirrors format_basis.
    pub fn format_ecp(
        input: &EcpInput,
        atoms: &[ParsedAtom],
    ) -> Result<HashMap<String, ParsedEcp>, PyscfRsError> {
        if matches!(input, EcpInput::None) {
            return Ok(HashMap::new());
        }
        // Same first-occurrence-symbol pattern as format_basis (Pitfall 4 alignment).
        let mut seen: Vec<String> = Vec::new();
        for (s, _) in atoms {
            let alpha: String = s.chars().take_while(|c| c.is_alphabetic()).collect();
            let upper = alpha.to_ascii_uppercase();
            if !seen.contains(&upper) { seen.push(upper); }
        }
        let mut out: HashMap<String, ParsedEcp> = HashMap::new();
        for symbol in &seen {
            if let Some(parsed) = resolve_for_symbol(input, symbol)? {
                out.insert(symbol.clone(), parsed);
            }
            // Symbols with no ECP entry are simply absent (per upstream).
        }
        Ok(out)
    }

    fn resolve_for_symbol(input: &EcpInput, symbol: &str) -> Result<Option<ParsedEcp>, PyscfRsError> {
        match input {
            EcpInput::None => Ok(None),
            EcpInput::Name(name) => {
                // ECP files in pyscf/gto/basis/ contain BOTH basis blocks AND ECP blocks
                // (e.g. lanl2dz.dat has H..Cs basis sections + Cu, Mn, ... ECP sections).
                // Resolve via the same ALIAS table; load file; pass to parse_nwchem_ecp.
                let canonical = basis::canonicalise_basis_name(name);
                let filename = match basis::alias::lookup(&canonical) {
                    Some(f) => f,
                    None => return Err(PyscfRsError::from(BasisLoadError::UnknownName {
                        name: format!("ECP basis '{}' not in ALIAS", canonical),
                    })),
                };
                let dir = basis::path::basis_dir().map_err(PyscfRsError::from)?;
                let full = dir.join(filename);
                let text = std::fs::read_to_string(&full).map_err(|e| {
                    PyscfRsError::from(BasisLoadError::Io {
                        path: full.display().to_string(), source: e,
                    })
                })?;
                // Look for "ECP" section in text; if not present → no ECP for this symbol.
                if !text.contains("ECP") {
                    return Ok(None);
                }
                match crate::basis::nwchem_ecp::parse_nwchem_ecp(&text, symbol, &full.display().to_string()) {
                    Ok(p) if p.channels.is_empty() => Ok(None),  // file has ECP for other elements but not ours
                    Ok(p) => Ok(Some(p)),
                    Err(e) => Err(PyscfRsError::from(e)),
                }
            }
            EcpInput::PerElement(map) => {
                let sub = map.get(symbol)
                    .or_else(|| map.get(&symbol.to_ascii_lowercase()))
                    .or_else(|| map.get("default"));
                match sub {
                    Some(s) => resolve_for_symbol(s, symbol),
                    None    => Ok(None),                        // no ECP for this symbol; not an error
                }
            }
            EcpInput::NwchemEcpText(text) => {
                let p = crate::basis::nwchem_ecp::parse_nwchem_ecp(text, symbol, "<inline-ecp-text>")
                    .map_err(PyscfRsError::from)?;
                if p.channels.is_empty() { Ok(None) } else { Ok(Some(p)) }
            }
            EcpInput::Parsed(p) => Ok(Some(p.clone())),
        }
    }

    /// make_ecp_env — projection of ECP data to flat arrays, mutating _atm + _env + _ecpbas.
    /// Source: pyscf/gto/mole.py:1107-1160. Pitfall 2: CHARGE_OF subtraction happens HERE
    /// (and only here).
    pub fn make_ecp_env(
        atoms: &[ParsedAtom],
        ecp: &HashMap<String, ParsedEcp>,
        _atm: &mut Vec<i32>,
        _env: &mut Vec<f64>,
    ) -> Vec<i32> {
        let mut _ecpbas: Vec<i32> = Vec::new();
        for (atom_id, (sym_with_suffix, _)) in atoms.iter().enumerate() {
            let alpha: String = sym_with_suffix.chars().take_while(|c| c.is_alphabetic()).collect();
            let upper = alpha.to_ascii_uppercase();
            let parsed = match ecp.get(&upper) { Some(p) => p, None => continue };

            // Pitfall 2: subtract n_core from CHARGE_OF (once, here).
            let row_start = atom_id * ATM_SLOTS;
            _atm[row_start + CHARGE_OF] -= parsed.n_core as i32;
            // Note: we do NOT change NUC_MOD_OF — v1 keeps POINT_NUC=1 for ECP atoms.
            // (Some references use NUC_MOD_OF=2 for ECP, but cintx convention follows
            // upstream PySCF which keeps NUC_MOD_OF independent of ECP presence; the
            // ECP itself is signalled by _ecpbas + the int1e_ecp* intor name.)

            // Append ECP shells to _ecpbas + ECP exponents/coeffs to _env.
            for ch in &parsed.channels {
                let ptr_exp = _env.len();
                _env.extend_from_slice(&ch.exponents);
                let ptr_coeff = _env.len();
                _env.extend_from_slice(&ch.coeffs);
                // Also append n_powers as f64 if upstream interleaves them (consult mole.py:1140+).
                // Upstream stores n_powers separately in env; verify by reading the line and adjust.

                let mut row = [0_i32; BAS_SLOTS];
                row[ATOM_OF]   = atom_id as i32;
                row[ANG_OF]    = ch.l as i32;
                row[NPRIM_OF]  = ch.exponents.len() as i32;
                row[NCTR_OF]   = 1;                         // ECP shells are 1-contraction
                row[KAPPA_OF]  = 0;
                row[PTR_EXP]   = ptr_exp as i32;
                row[PTR_COEFF] = ptr_coeff as i32;
                _ecpbas.extend_from_slice(&row);
            }
        }
        debug_assert_eq!(_ecpbas.len() % BAS_SLOTS, 0, "Pitfall 2: _ecpbas dim");
        _ecpbas
    }
