    ```rust
    //! GTO-07 + D-04: cubecl AO-on-grid kernel.
    //!
    //! Source: pyscf/gto/eval_gto.py (Apache-2.0). Algorithm: per-grid-point thread
    //! evaluates each shell's contracted radial × cart2sph transform.
    //!
    //! Algebra-wall: this module imports cubecl directly (allowed via Wave 0 W0-T4
    //! allowlist update). The PUBLIC function `eval_gto_sph` takes only pyscf-algebra
    //! types — pyscf-gto's wrapper imports this without ever naming cubecl.

    use cubecl::prelude::*;
    use cubecl_cpu::CpuRuntime;       // default backend; per-feature dispatch added later
    use pyscf_algebra::{AlgebraClient, BackendKind, Tensor};

    /// Output of an eval_gto call.
    #[derive(Debug)]
    pub struct EvalGtoBuffers {
        /// Output buffer: shape (ngrids, nao) for scalar variants, (3, ngrids, nao) for ip/ig.
        /// F-order on flatten — matches upstream PySCF pyscf/gto/eval_gto.py output.
        pub values: Vec<f64>,
        pub shape: Vec<usize>,
    }

    /// Evaluate `GTOval_sph` (or GTOval_cart) on the given grid for the supplied basis.
    /// Public surface uses pyscf-algebra types only — pyscf-gto's wrapper calls this.
    ///
    /// Inputs:
    ///   coords: flat Vec<f64> of length ngrids*3 (F-order: x[0..ngrids], y[ngrids..2*ngrids], z[..])
    ///   atm/bas/env/ao_loc: the libcint flat arrays from mol._atm/_bas/_env/ao_loc_nr
    ///   nao: total number of AOs (mol.nao_nr)
    ///   spherical: true → apply cart2sph; false → return raw cartesian
    pub fn eval_gto_sph(
        client: &AlgebraClient,
        coords: &[f64], ngrids: usize,
        atm: &[i32], bas: &[i32], env: &[f64], ao_loc: &[i32],
        nao: usize,
        spherical: bool,
    ) -> EvalGtoBuffers {
        // Match on backend; for Phase 2 ship CPU only. Cuda/Wgpu/Rocm covered by feature
        // gating but route to a not-yet-implemented panic guard until Phase 8.
        match client.kind() {
            BackendKind::Cpu => eval_gto_sph_cpu(coords, ngrids, atm, bas, env, ao_loc, nao, spherical),
            other => {
                tracing::warn!(backend = ?other, "eval_gto: backend not yet wired; falling back to CPU");
                eval_gto_sph_cpu(coords, ngrids, atm, bas, env, ao_loc, nao, spherical)
            }
        }
    }

    fn eval_gto_sph_cpu(
        coords_host: &[f64], ngrids: usize,
        atm_host: &[i32], bas_host: &[i32], env_host: &[f64], ao_loc_host: &[i32],
        nao: usize,
        spherical: bool,
    ) -> EvalGtoBuffers {
        let cube_client = CpuRuntime::client(&Default::default());

        let coords  = cube_client.create(bytemuck::cast_slice(coords_host));
        let atm     = cube_client.create(bytemuck::cast_slice(atm_host));
        let bas     = cube_client.create(bytemuck::cast_slice(bas_host));
        let env     = cube_client.create(bytemuck::cast_slice(env_host));
        let ao_loc  = cube_client.create(bytemuck::cast_slice(ao_loc_host));
        let out_len = ngrids * nao;
        let out     = cube_client.empty(out_len * std::mem::size_of::<f64>());

        let nbas = bas_host.len() / pyscf_core::raw_layout::BAS_SLOTS;
        let cube_dim_x = 32_u32;
        let cube_count = ((ngrids as u32 + cube_dim_x - 1) / cube_dim_x).max(1);

        unsafe {
            eval_gto_sph_kernel::launch_unchecked::<CpuRuntime>(
                &cube_client,
                CubeCount::Static(cube_count, 1, 1),
                CubeDim::new_1d(cube_dim_x),
                ArrayArg::from_raw_parts::<f64>(&coords, ngrids * 3, 1),
                ArrayArg::from_raw_parts::<i32>(&atm,    atm_host.len(), 1),
                ArrayArg::from_raw_parts::<i32>(&bas,    bas_host.len(), 1),
                ArrayArg::from_raw_parts::<f64>(&env,    env_host.len(), 1),
                ArrayArg::from_raw_parts::<i32>(&ao_loc, ao_loc_host.len(), 1),
                ArrayArg::from_raw_parts::<f64>(&out,    out_len, 1),
                ngrids as u32,
                nao    as u32,
                nbas   as u32,
                spherical,
            );
        }

        let raw = cube_client.read_one(out.binding());
        let values: Vec<f64> = bytemuck::cast_slice::<u8, f64>(&raw).to_vec();
        EvalGtoBuffers { values, shape: vec![ngrids, nao] }
    }

    /// Per-grid-point AO-on-grid kernel.
    /// One thread per grid point evaluates ALL shells on that point.
    /// Output: F-order (ngrids, nao) — index [g + ao * ngrids].
    ///
    /// Phase 2 MVP shape: a Phase 4 DFT-ready stub that handles l = 0 (s shells) only.
    /// l ≥ 1 (p, d, f) and the cart2sph transform are documented as Phase 4 DFT
    /// extensions — `GTOval_sph` for s shells alone covers the smoke test
    /// (H 1s at nucleus = analytical contracted radial value).
    ///
    /// Phase 4 DFT extends this kernel with: full cart2sph for l = 1..4, ip/ig
    /// derivative variants, deriv1/deriv2 stencils.
    #[cube(launch_unchecked)]
    fn eval_gto_sph_kernel(
        coords: &Array<f64>,        // ngrids * 3 (F-order: x[0..ngrids], y[ngrids..], z[2*ngrids..])
        atm:    &Array<i32>,        // natm * ATM_SLOTS=6
        bas:    &Array<i32>,        // nbas * BAS_SLOTS=8
        env:    &Array<f64>,
        ao_loc: &Array<i32>,        // nbas+1
        out:    &mut Array<f64>,    // ngrids * nao (F-order: out[g + ao * ngrids])
        #[comptime] ngrids: u32,
        #[comptime] nao:    u32,
        #[comptime] nbas:   u32,
        #[comptime] _spherical: bool,
    ) {
        // Slot constants — must match cintx_compat::raw exactly. Inlined here because
        // #[cube] kernels can't reference Rust constants from outside the macro.
        // pyscf-rs CI lints (Phase 1 dependency-wall; Phase 2 byte-identity oracle)
        // catch any drift.
        let ATM_SLOTS: u32 = 6;
        let BAS_SLOTS: u32 = 8;
        let CHARGE_OF: u32 = 0;
        let PTR_COORD: u32 = 1;
        let _NUC_MOD_OF: u32 = 2;
        let ATOM_OF:   u32 = 0;
        let ANG_OF:    u32 = 1;
        let NPRIM_OF:  u32 = 2;
        let NCTR_OF:   u32 = 3;
        let _KAPPA_OF: u32 = 4;
        let PTR_EXP:   u32 = 5;
        let PTR_COEFF: u32 = 6;

        let g = ABSOLUTE_POS;
        if g >= ngrids { return; }

        let gx = coords[g];
        let gy = coords[g + ngrids];
        let gz = coords[g + 2 * ngrids];

        let mut shell_idx: u32 = 0;
        loop {
            if shell_idx >= nbas { break; }
            let bas_row    = shell_idx * BAS_SLOTS;
            let atom_id    = bas[bas_row + ATOM_OF] as u32;
            let l          = bas[bas_row + ANG_OF] as u32;
            let nprim      = bas[bas_row + NPRIM_OF] as u32;
            let nctr       = bas[bas_row + NCTR_OF] as u32;
            let ptr_exp    = bas[bas_row + PTR_EXP] as u32;
            let ptr_coeff  = bas[bas_row + PTR_COEFF] as u32;

            let atm_row    = atom_id * ATM_SLOTS;
            let ptr_coord  = atm[atm_row + PTR_COORD] as u32;
            let ax = env[ptr_coord];
            let ay = env[ptr_coord + 1];
            let az = env[ptr_coord + 2];

            let dx = gx - ax;
            let dy = gy - ay;
            let dz = gz - az;
            let r2 = dx * dx + dy * dy + dz * dz;

            // l == 0 path (s shell): contracted radial × Y_00 = contracted_radial × 1
            // (Y_00 = 1/(2*sqrt(π)) is absorbed into the normalised coefficients.)
            // For Phase 2, only l == 0 path is wired; l ≥ 1 yields output 0.0 and the
            // smoke test asserts the s-shell value. Phase 4 DFT extends.
            if l == 0 {
                let mut c_idx: u32 = 0;
                loop {
                    if c_idx >= nctr { break; }
                    let mut acc: f64 = 0.0;
                    let mut p_idx: u32 = 0;
                    loop {
                        if p_idx >= nprim { break; }
                        let alpha = env[ptr_exp + p_idx];
                        // Coefficient matrix is F-order: ptr_coeff + c_idx * nprim + p_idx
                        let coef = env[ptr_coeff + c_idx * nprim + p_idx];
                        acc = acc + coef * f64::exp(-alpha * r2);
                        p_idx = p_idx + 1;
                    }
                    // Output index: F-order (ngrids, nao); ao_idx = ao_loc[shell_idx] + c_idx
                    let ao_idx = ao_loc[shell_idx] as u32 + c_idx;
                    out[g + ao_idx * ngrids] = acc;
                    c_idx = c_idx + 1;
                }
            } else {
                // l >= 1 path: stub for Phase 2 — write zeros. Phase 4 DFT plan extends.
                let dim_per_ctr: u32 = 2 * l + 1;          // sph
                let mut c_idx: u32 = 0;
                loop {
                    if c_idx >= nctr { break; }
                    let mut m_idx: u32 = 0;
                    loop {
                        if m_idx >= dim_per_ctr { break; }
                        let ao_idx = ao_loc[shell_idx] as u32 + c_idx * dim_per_ctr + m_idx;
                        out[g + ao_idx * ngrids] = 0.0;
                        m_idx = m_idx + 1;
                    }
                    c_idx = c_idx + 1;
                }
            }

            shell_idx = shell_idx + 1;
        }
    }

    /// Stub for derivative variants. Phase 4 DFT or Phase 7 grad extends.
    pub fn eval_gto_unsupported(name: &str) -> EvalGtoBuffers {
        // Caller checks for empty `values` and converts to NotYetImplemented error.
        let _ = name;
        EvalGtoBuffers { values: Vec::new(), shape: vec![0, 0] }
    }
    ```

    ```rust
    //! GTO-07: eval_gto user wrapper. Algebra-wall friendly — no cubecl import.
    //! Source: pyscf/gto/eval_gto.py (Apache-2.0).

    use pyscf_algebra::AlgebraClient;
    use pyscf_core::{CoreError, Mole, PyscfRsError};

    /// Output of an eval_gto call.
    #[derive(Debug, Clone)]
    pub struct EvalGtoOutput {
        /// Flat F-order buffer. For scalar variants (`GTOval`, `GTOval_sph`, `GTOval_cart`):
        /// index `g + ao * ngrids`. For derivative variants (Phase 4): leading axis = component.
        pub values: Vec<f64>,
        pub shape: Vec<usize>,
    }

    /// Evaluate AO values at the given grid coordinates.
    /// `coords` shape: (ngrids, 3) C-order — `coords[g*3 + xyz]`.
    /// 02-06 supports `GTOval`, `GTOval_sph`, `GTOval_cart`. Derivative variants
    /// return `NotYetImplemented{phase:4}`.
    pub fn eval_gto(mol: &Mole, eval_name: &str, coords: &[[f64; 3]]) -> Result<EvalGtoOutput, PyscfRsError> {
        if !mol._built {
            return Err(PyscfRsError::Core(CoreError::InvalidMolecule(
                "Mole not built — call mol.build() first".into(),
            )));
        }

        // Resolve alias `GTOval` to `GTOval_sph` or `GTOval_cart` per mol.cart.
        let resolved = match eval_name {
            "GTOval" => if mol.cart { "GTOval_cart" } else { "GTOval_sph" },
            other    => other,
        };

        // Variant dispatch.
        match resolved {
            "GTOval_sph" | "GTOval_cart" => {
                let spherical = resolved == "GTOval_sph";
                let ngrids = coords.len();

                // Pack coords into F-order flat buffer: x[0..ngrids], y[ngrids..2N], z[..3N].
                let mut coords_flat = Vec::with_capacity(ngrids * 3);
                coords_flat.extend(coords.iter().map(|c| c[0]));
                coords_flat.extend(coords.iter().map(|c| c[1]));
                coords_flat.extend(coords.iter().map(|c| c[2]));

                let client = AlgebraClient::auto();   // PYSCF_BACKEND-driven
                let buffers = pyscf_kernels::eval_gto_sph(
                    &client,
                    &coords_flat, ngrids,
                    &mol._atm, &mol._bas, &mol._env, &mol.ao_loc_nr,
                    mol.nao_nr,
                    spherical,
                );
                Ok(EvalGtoOutput { values: buffers.values, shape: buffers.shape })
            }
            "GTOval_sph_deriv1" | "GTOval_cart_deriv1" => Err(PyscfRsError::NotYetImplemented {
                phase: 4, what: "GTOval_sph_deriv1 (Phase 4 DFT grid integration extends the kernel)",
            }),
            "GTOval_sph_deriv2" | "GTOval_cart_deriv2" => Err(PyscfRsError::NotYetImplemented {
                phase: 4, what: "GTOval_sph_deriv2",
            }),
            "GTOval_ip_sph" | "GTOval_ip_cart" | "GTOval_ip" => Err(PyscfRsError::NotYetImplemented {
                phase: 7, what: "GTOval_ip (gradient variant; Phase 7 grad)",
            }),
            "GTOval_ig_sph" | "GTOval_ig_cart" | "GTOval_ig" => Err(PyscfRsError::NotYetImplemented {
                phase: 7, what: "GTOval_ig (magnetic gauge; Phase 7 grad)",
            }),
            other => Err(PyscfRsError::Core(CoreError::InvalidMolecule(
                format!("unknown eval_gto variant: {} (Phase 2 ships GTOval, GTOval_sph, GTOval_cart; deriv variants stub to Phase 4/7)", other)
            ))),
        }
    }
    ```
