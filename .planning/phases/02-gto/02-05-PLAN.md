    ```rust
    //! GTO-06: mol.intor(name) — dispatcher over cintx_rs::SessionRequest.
    //!
    //! Source: pyscf/gto/mole.py:945+ (suffix appending) + pyscf/gto/moleintor.py:41-271
    //! (getints dispatcher) — pyscf-rs delegates the heavy lifting to cintx-rs.
    //!
    //! Pitfall 1: per-intor F/C-order layout consultation via layout_table::lookup.

    use crate::layout_table::{self, IntorLayout};
    use cintx_core::{OperatorId, Representation, ShellTuple};
    use cintx_rs::{ExecutionOptions, SessionRequest};
    use pyscf_core::{CoreError, Mole, PyscfRsError};

    /// Output of a successful intor call. The caller reshapes via `shape` + `layout`.
    /// `values` is F-order (component axis leading where applicable).
    #[derive(Debug, Clone)]
    pub struct IntorOutput {
        pub values: Vec<f64>,
        pub shape: Vec<usize>,
        pub layout: IntorLayout,
    }

    /// Compute the named integral. Source: pyscf/gto/mole.py:945+ + moleintor.py:41+.
    pub fn intor(mol: &Mole, name: &str) -> Result<IntorOutput, PyscfRsError> {
        // Built check.
        if !mol._built {
            return Err(PyscfRsError::Core(CoreError::InvalidMolecule(
                "Mole not built — call mol.build() first".into(),
            )));
        }

        // Suffix normalisation per upstream _add_suffix.
        let full_name = add_suffix(name, mol.cart);

        // ECP route per Phase 2 D-07: int1e_ecp* and ECPscalar* go to EcpEngine.
        if full_name.starts_with("int1e_ecp") || full_name.starts_with("ECPscalar") {
            // 02-07 wires the actual trait dispatch; 02-05 returns the stub error directly
            // so this plan ships independently of 02-07.
            return Err(PyscfRsError::EcpEngineNotAvailable);
        }

        // Layout lookup (Pitfall 1 — per-intor F/C-order).
        let layout = layout_table::lookup(&full_name).ok_or_else(|| {
            PyscfRsError::Core(CoreError::InvalidMolecule(
                format!("unknown intor: {} (not in INTOR_LAYOUTS — Phase 2 catalogue covers {} entries)",
                    full_name, crate::layout_table::INTOR_LAYOUTS.len())
            ))
        })?;

        // Get cintx basis (Arc; zero-copy per GTO-11).
        let basis_arc = mol.cintx_basis()?;
        let basis_ref: &cintx_core::BasisSet = &basis_arc;

        // Construct shells covering all shells in the basis.
        // Read cintx_core::ShellTuple for the canonical builder; the wave0 smoke test
        // already showed the construction pattern. Here we want "all shells".
        let nbas = mol.nbas;
        let shells = build_full_shells(basis_ref, nbas)?;

        // Choose Representation per intor name suffix (sph / cart).
        // (mol.cart drives the suffix; the suffix tells cintx which Representation to use.)
        let representation = if full_name.ends_with("_cart") {
            Representation::Cart
        } else if full_name.ends_with("_sph") {
            Representation::Spheric
        } else {
            // Spinor not in v1 scope; Phase 3 may extend.
            return Err(PyscfRsError::NotYetImplemented {
                phase: 3, what: "spinor representation",
            });
        };

        // Resolve cintx OperatorId.
        let operator = OperatorId::for_symbol(&full_name);   // verify by reading cintx_core::operator
        // If for_symbol returns Option, unwrap with a useful error here.

        let request = SessionRequest::new(
            operator,
            representation,
            basis_ref,
            shells,
            ExecutionOptions::default(),
        );
        let outcome = request.query_workspace()
            .map_err(|e| PyscfRsError::Core(CoreError::InvalidMolecule(
                format!("cintx workspace query failed for {}: {:?}", full_name, e)
            )))?
            .evaluate()
            .map_err(|e| PyscfRsError::Core(CoreError::InvalidMolecule(
                format!("cintx evaluate failed for {}: {:?}", full_name, e)
            )))?;

        // Reshape per upstream F-order convention (Pitfall 1).
        // For ScalarFOrder layouts: shape = extents as-is (cintx already returns F-order).
        // For ComponentLeadingFOrder: shape[0] is component count; layout flag tells caller.
        let shape = outcome.tensor.extents.clone();
        Ok(IntorOutput {
            values: outcome.tensor.owned_values,
            shape,
            layout,
        })
    }

    /// Append `_sph` / `_cart` suffix per upstream _add_suffix at pyscf/gto/mole.py:945+.
    fn add_suffix(intor: &str, cart: bool) -> String {
        if intor.ends_with("_sph") || intor.ends_with("_cart") || intor.ends_with("_spinor") {
            return intor.to_string();
        }
        if cart { format!("{}_cart", intor) } else { format!("{}_sph", intor) }
    }

    /// Build the all-shells ShellTuple cintx needs for full-basis intors.
    /// Read cintx/crates/cintx-core/src/shell.rs for the canonical builder.
    fn build_full_shells(basis: &cintx_core::BasisSet, nbas: usize) -> Result<ShellTuple, PyscfRsError> {
        // Wave 0 02-01 already constructed this; lift the same pattern.
        // Placeholder body — actual API depends on cintx_core::shell exports.
        // The cintx tests/examples are the authoritative reference.
        let shells: Vec<usize> = (0..nbas).collect();
        // Construct ShellTuple from the shell-index list.
        // If cintx_core::ShellTuple::from_indices(basis, shells) exists, use that.
        // Otherwise consult cintx-rs/tests for the construction.
        ShellTuple::from_indices(basis, &shells)
            .map_err(|e| PyscfRsError::Core(CoreError::InvalidMolecule(
                format!("ShellTuple::from_indices failed for nbas={}: {:?}", nbas, e)
            )))
    }

    #[cfg(test)]
    mod tests {
        use super::*;

        #[test]
        fn add_suffix_appends_sph_by_default() {
            assert_eq!(add_suffix("int1e_ovlp", false), "int1e_ovlp_sph");
        }
        #[test]
        fn add_suffix_appends_cart_when_set() {
            assert_eq!(add_suffix("int1e_ovlp", true), "int1e_ovlp_cart");
        }
        #[test]
        fn add_suffix_passes_through_already_suffixed() {
            assert_eq!(add_suffix("int1e_ovlp_sph", false), "int1e_ovlp_sph");
            assert_eq!(add_suffix("int1e_ovlp_cart", true), "int1e_ovlp_cart");
            assert_eq!(add_suffix("int1e_ovlp_spinor", false), "int1e_ovlp_spinor");
        }
    }
    ```
