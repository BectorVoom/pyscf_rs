    (a) Create `tests/oracle/test_byte_identity.py`:
    ```python
    """GTO-04: byte-identity of _atm/_bas/_env/ao_loc_nr/nao_nr vs upstream PySCF.

    Per Phase 2 success criterion 1 (ROADMAP). Pitfall 17 mitigation: the
    `ao_loc_nr` byte-equal assertion catches off-by-one in basis indexing.
    """
    import json
    import os
    import subprocess
    import numpy as np
    import pytest

    PR_CI_FIXTURES = [
        ("h2o_ccpvdz",     "O 0 0 0; H 0 0.7 0.6; H 0 -0.7 0.6", "cc-pvdz"),
        ("benzene_631gs",  # benzene at experimental geometry (D6h)
         "C 0 1.397 0; C 1.21 0.6985 0; C 1.21 -0.6985 0; "
         "C 0 -1.397 0; C -1.21 -0.6985 0; C -1.21 0.6985 0; "
         "H 0 2.481 0; H 2.149 1.2405 0; H 2.149 -1.2405 0; "
         "H 0 -2.481 0; H -2.149 -1.2405 0; H -2.149 1.2405 0",
         "6-31g*"),
        ("water_trimer_sto3g",
         "O 0 0 0; H 0 0.7 0.6; H 0 -0.7 0.6; "
         "O 3 0 0; H 3.7 0.6 0; H 2.3 0.6 0; "
         "O -3 0 0; H -3.7 0.6 0; H -2.3 0.6 0",
         "sto-3g"),
    ]

    def _upstream_arrays(upstream_pyscf, atom, basis):
        mol = upstream_pyscf.M(atom=atom, basis=basis, unit="Bohr", verbose=0)
        return {
            "_atm":      np.asarray(mol._atm, dtype=np.int32).flatten(),
            "_bas":      np.asarray(mol._bas, dtype=np.int32).flatten(),
            "_env":      np.asarray(mol._env, dtype=np.float64),
            "ao_loc_nr": np.asarray(mol.ao_loc_nr(), dtype=np.int32),
            "nao_nr":    int(mol.nao_nr()),
        }

    def _pyscf_rs_arrays(name, atom, basis):
        """Invoke a pyscf-rs CLI/test helper that dumps the arrays to a JSON file.
        Returns the parsed JSON arrays.
        """
        out_path = f"tests/oracle/.tmp/{name}_pyscfrs.json"
        os.makedirs(os.path.dirname(out_path), exist_ok=True)
        env = os.environ.copy()
        env.update({
            "PYSCF_RS_ORACLE_ATOM": atom,
            "PYSCF_RS_ORACLE_BASIS": basis,
            "PYSCF_RS_ORACLE_OUT":   out_path,
        })
        result = subprocess.run(
            ["cargo", "test", "--release", "--features", "release-oracle-tests",
             "-p", "pyscf-gto", "--test", "dump_arrays_for_oracle",
             "--", "--nocapture", "--ignored"],
            env=env, capture_output=True, text=True,
        )
        assert result.returncode == 0, f"cargo test failed: {result.stderr[:2000]}"
        with open(out_path) as f:
            data = json.load(f)
        return {
            "_atm":      np.asarray(data["_atm"],      dtype=np.int32),
            "_bas":      np.asarray(data["_bas"],      dtype=np.int32),
            "_env":      np.asarray(data["_env"],      dtype=np.float64),
            "ao_loc_nr": np.asarray(data["ao_loc_nr"], dtype=np.int32),
            "nao_nr":    int(data["nao_nr"]),
        }

    @pytest.mark.parametrize("name,atom,basis", PR_CI_FIXTURES)
    def test_atm_bas_env_byte_for_byte(upstream_pyscf, name, atom, basis):
        u = _upstream_arrays(upstream_pyscf, atom, basis)
        r = _pyscf_rs_arrays(name, atom, basis)
        np.testing.assert_array_equal(r["_atm"], u["_atm"], err_msg=f"_atm mismatch for {name}")
        np.testing.assert_array_equal(r["_bas"], u["_bas"], err_msg=f"_bas mismatch for {name}")
        # _env: f64 — bit-equal under release-oracle (Phase 1 FMA-free profile).
        np.testing.assert_array_equal(r["_env"], u["_env"], err_msg=f"_env mismatch for {name}")

    @pytest.mark.parametrize("name,atom,basis", PR_CI_FIXTURES)
    def test_ao_loc_nr_byte_for_byte(upstream_pyscf, name, atom, basis):
        """Pitfall 17 mitigation: off-by-one basis indexing → ao_loc_nr drift."""
        u = _upstream_arrays(upstream_pyscf, atom, basis)
        r = _pyscf_rs_arrays(name, atom, basis)
        np.testing.assert_array_equal(
            r["ao_loc_nr"], u["ao_loc_nr"],
            err_msg=f"ao_loc_nr mismatch for {name} (Pitfall 17 — off-by-one in basis indexing)"
        )
        assert r["nao_nr"] == u["nao_nr"], \
            f"nao_nr mismatch for {name}: {r['nao_nr']} vs {u['nao_nr']}"
    ```

    ```python
    """GTO-06: mol.intor parity vs upstream PySCF.

    Pitfall 1 (F-order layout): test_int1e_ipovlp_sph_layout asserts the
    component-leading 3D output preserves shape (3, nao, nao) byte-equal to upstream.
    """
    import os
    import subprocess
    import json
    import numpy as np
    import pytest

    INTORS_TO_VERIFY = [
        "int1e_ovlp_sph",
        "int1e_kin_sph",
        "int1e_nuc_sph",
        "int2e_sph",
        "int1e_ipovlp_sph",   # Pitfall 1: component-leading (3, nao, nao)
        "int1e_ipnuc_sph",
        "int1e_iprinv_sph",
        "int3c2e_sph",
        "int2c2e_sph",
        "int1e_r_sph",        # 3-component dipole
    ]

    def _intor_upstream(mol, name):
        return mol.intor(name)        # numpy F-order

    def _intor_pyscfrs(name, atom, basis, intor_name):
        out_path = f"tests/oracle/.tmp/{name}_intor_{intor_name}.json"
        os.makedirs(os.path.dirname(out_path), exist_ok=True)
        env = os.environ.copy()
        env.update({
            "PYSCF_RS_ORACLE_ATOM": atom, "PYSCF_RS_ORACLE_BASIS": basis,
            "PYSCF_RS_ORACLE_INTOR": intor_name, "PYSCF_RS_ORACLE_OUT": out_path,
        })
        result = subprocess.run(
            ["cargo", "test", "--release", "--features", "release-oracle-tests",
             "-p", "pyscf-gto", "--test", "dump_intor_for_oracle",
             "--", "--ignored"],
            env=env, capture_output=True, text=True,
        )
        assert result.returncode == 0, f"cargo test failed: {result.stderr[:2000]}"
        with open(out_path) as f:
            data = json.load(f)
        return np.asarray(data["values"], dtype=np.float64).reshape(data["shape"], order="F")

    @pytest.mark.parametrize("intor_name", INTORS_TO_VERIFY)
    def test_intor_h2o_ccpvdz(upstream_pyscf, intor_name):

        basis = "cc-pvdz"
        mol = upstream_pyscf.M(atom=atom, basis=basis, unit="Bohr", verbose=0)
        u = _intor_upstream(mol, intor_name)
        r = _intor_pyscfrs("h2o_ccpvdz", atom, basis, intor_name)
        # Bit-equal under release-oracle profile (FMA-free); cintx tolerance otherwise.
        # Phase 2 oracle policy: chemical accuracy (1e-10) per the cintx contract.
        np.testing.assert_allclose(r, u, atol=1e-10, rtol=1e-10,
            err_msg=f"intor {intor_name} mismatch on H2O/cc-pvdz")

    def test_int1e_ipovlp_sph_layout(upstream_pyscf):
        """Pitfall 1: component-leading 3D layout preserved."""
        atom = "H 0 0 0; H 0 0 1.4"
        basis = "sto-3g"
        mol = upstream_pyscf.M(atom=atom, basis=basis, unit="Bohr", verbose=0)
        u = mol.intor("int1e_ipovlp_sph")
        assert u.shape == (3, 2, 2), f"upstream shape {u.shape} (expected (3,2,2))"
        # Just verifying upstream's expected shape; the byte-identity check is in
        # test_intor_h2o_ccpvdz parametric run.
    ```

    (d) Create `tests/oracle/test_json_interop.py`:
    ```python
    """GTO-09 cross-language: pyscf-rs writes JSON; Python (upstream) reads;
    re-build via upstream `pyscf.M(**state)` produces equivalent Mole arrays.

    Per CONTEXT D-09: semantic round-trip ONLY (not byte-identical to upstream JSON).
    The contract is: state preserved → re-build reconstructs.
    """
    import os
    import subprocess
    import json
    import numpy as np
    import pytest

    def _pyscf_rs_dump(name, atom, basis):
        out_path = f"tests/oracle/.tmp/{name}_dumps.json"
        os.makedirs(os.path.dirname(out_path), exist_ok=True)
        env = os.environ.copy()
        env.update({
            "PYSCF_RS_ORACLE_ATOM": atom, "PYSCF_RS_ORACLE_BASIS": basis,
            "PYSCF_RS_ORACLE_OUT": out_path,
        })
        result = subprocess.run(
            ["cargo", "test", "--release", "--features", "release-oracle-tests",
             "-p", "pyscf-gto", "--test", "dump_mole_dumps_for_oracle",
             "--", "--ignored"],
            env=env, capture_output=True, text=True,
        )
        assert result.returncode == 0, f"cargo test failed: {result.stderr[:2000]}"
        with open(out_path) as f:
            return f.read()

    def test_pyscfrs_dumps_to_pyscf_loads_roundtrip(upstream_pyscf):
        """Semantic round-trip: pyscf-rs writes JSON; we re-read user inputs and
        feed back into upstream pyscf.M(); the resulting Mole's _atm matches.
        """
        atom = "H 0 0 0; H 0 0 1.4"
        basis = "sto-3g"
        json_str = _pyscf_rs_dump("h2_sto3g", atom, basis)
        snap = json.loads(json_str)
        # Reconstruct via upstream from the snapshot's atom_parsed + basis_kind.Name.
        rebuilt_atoms = [(s, [c[0], c[1], c[2]]) for s, c in snap["atom_parsed"]]
        # Match basis kind.
        basis_repr = snap["basis_kind"]
        if "Name" in basis_repr:
            basis_name = basis_repr["Name"]
        else:
            pytest.skip("Phase 2 only round-trips Name-form basis through this test")
    ```
