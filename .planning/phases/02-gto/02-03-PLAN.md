    ```rust
    //! GTO-02 + GTO-03 basis-set loading.
    //!
    //! D-01: Runtime resolution + lazy parse + OnceLock cache. No build.rs codegen.
    //! D-02: PYSCF_BASIS_PATH env override → CARGO_MANIFEST_DIR walk-up → error.
    //!
    //! Layout mirrors upstream pyscf/gto/basis/ (4 parser modules per Discretion).

    pub mod path;
    pub mod alias;
    pub mod nwchem;
    pub mod nwchem_ecp;
    pub mod cp2k;
    pub mod cp2k_pp;

    use pyscf_core::{BasisLoadError, ParsedBasis};
    use std::collections::HashMap;
    use std::sync::{Mutex, OnceLock};

    /// Process-local parsed-basis cache. Key: `(canonical_basis_name, element_symbol)`.
    /// Lookups are protected by a `Mutex` (per RESEARCH Pitfall 6 — simpler than the
    /// per-name OnceLock-in-RwLock pattern; basis load is dwarfed by integral eval).
    static BASIS_CACHE: OnceLock<Mutex<HashMap<(String, String), ParsedBasis>>> = OnceLock::new();

    fn cache() -> &'static Mutex<HashMap<(String, String), ParsedBasis>> {
        BASIS_CACHE.get_or_init(|| Mutex::new(HashMap::new()))
    }

    /// Load a basis for one element symbol. First call parses + caches; subsequent
    /// calls hit the cache. Source: pyscf/gto/basis/__init__.py:621-728 `load`.
    pub fn load_basis(name: &str, symbol: &str) -> Result<ParsedBasis, BasisLoadError> {
        let canonical = canonicalise_basis_name(name);
        let key = (canonical.clone(), symbol.to_ascii_uppercase());

        // Fast path: cache hit.
        if let Some(parsed) = cache().lock().unwrap().get(&key) {
            return Ok(parsed.clone());
        }

        // Slow path: resolve filename via ALIAS, read file, dispatch parser.
        let filename = alias::lookup(&canonical)
            .ok_or_else(|| BasisLoadError::UnknownName { name: canonical.clone() })?;
        let dir = path::basis_dir()?;
        let full = dir.join(filename);
        let text = std::fs::read_to_string(&full).map_err(|e| BasisLoadError::Io {
            path: full.display().to_string(),
            source: e,
        })?;

        // Format detection: GTH → CP2K; otherwise NWChem.
        let parsed = if text.contains("GTH") {
            cp2k::parse_cp2k(&text, &key.1, &full.display().to_string())?
        } else {
            nwchem::parse_nwchem(&text, &key.1, &full.display().to_string())?
        };

        cache().lock().unwrap().insert(key.clone(), parsed.clone());
        tracing::debug!(name = %canonical, symbol = %key.1, file = %full.display(), "basis loaded");
        Ok(parsed)
    }

    /// Lower-case + strip "-" / "_" so "cc-pVDZ", "CC-PVDZ", "ccpvdz" all map to
    /// the same ALIAS key. Mirrors upstream `_format_basis_name` at
    /// pyscf/gto/basis/__init__.py:625.
    pub fn canonicalise_basis_name(name: &str) -> String {
        name.to_ascii_lowercase()
            .replace('-', "")
            .replace('_', "")
    }

    /// User entry point: parse Gaussian-94 / NWChem text directly (no file).
    /// Source: pyscf/gto/basis/__init__.py:730-779 `parse`.
    pub fn parse(text: &str, symbol: &str) -> Result<ParsedBasis, BasisLoadError> {
        if text.contains("GTH") {
            cp2k::parse_cp2k(text, symbol, "<inline-text>")
        } else {
            nwchem::parse_nwchem(text, symbol, "<inline-text>")
        }
    }
    ```

    ```rust
    //! D-02: PYSCF_BASIS_PATH resolver with priority chain.
    //!
    //! Priority:
    //! 1. PYSCF_BASIS_PATH env var (if set and is_dir)
    //! 2. CARGO_MANIFEST_DIR walk-up looking for `../../pyscf/gto/basis`
    //! 3. CARGO_MANIFEST_DIR walk-up looking for `../pyscf/gto/basis`
    //! 4. Error: BasisLoadError::PathNotFound

    use pyscf_core::BasisLoadError;
    use std::path::PathBuf;
    use std::sync::OnceLock;

    static BASIS_DIR: OnceLock<Option<PathBuf>> = OnceLock::new();

    pub fn basis_dir() -> Result<&'static PathBuf, BasisLoadError> {
        let resolved = BASIS_DIR.get_or_init(|| {
            // (1) env override
            if let Ok(p) = std::env::var("PYSCF_BASIS_PATH") {
                let path = PathBuf::from(&p);
                if path.is_dir() {
                    tracing::info!(path = %path.display(), "PYSCF_BASIS_PATH resolved basis dir");
                    return Some(path);
                }
                tracing::warn!(path = %p, "PYSCF_BASIS_PATH set but not a directory; falling through");
            }
            // (2) + (3) walk-up from manifest dir
            let candidates = [
                PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../../pyscf/gto/basis"),
                PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("../pyscf/gto/basis"),
                PathBuf::from(env!("CARGO_MANIFEST_DIR")).join("pyscf/gto/basis"),
            ];
            for c in &candidates {
                if c.is_dir() {
                    let canonical = c.canonicalize().unwrap_or_else(|_| c.clone());
                    tracing::info!(path = %canonical.display(), "walk-up resolved basis dir");
                    return Some(canonical);
                }
            }
            None
        });

        resolved.as_ref().ok_or_else(|| BasisLoadError::PathNotFound {
            tried: "PYSCF_BASIS_PATH env, CARGO_MANIFEST_DIR/../../pyscf/gto/basis, CARGO_MANIFEST_DIR/../pyscf/gto/basis".into(),
        })
    }

    #[cfg(test)]
    mod tests {
        use super::*;
        #[test]
        fn dev_walk_up_finds_pyscf_basis() {
            // The repo root has pyscf/gto/basis/ at top level; walk-up MUST find it.
            let dir = basis_dir().expect("walk-up should find pyscf/gto/basis");
            assert!(dir.is_dir(), "{:?} is not a dir", dir);
            // Sanity: the directory contains sto-3g.dat (a known builtin).
            assert!(dir.join("sto-3g.dat").is_file(),
                "expected {}/sto-3g.dat to exist", dir.display());
        }
    }
    ```

    ```rust
    //! NWChem / Gaussian-94 basis-set parser.
    //!
    //! Source: pyscf/gto/basis/parse_nwchem.py (Apache-2.0).
    //! Algorithm: line-stream parser; BASIS / END delimiters optional; element +
    //! angular-momentum-letter line opens a shell block; primitive rows follow;
    //! blank line OR new element-symbol-line closes the block.

    use pyscf_core::{BasisLoadError, ParsedBasis, ShellSpec};

    /// Parse NWChem / Gaussian-94 text for one element symbol.
    /// Returns `ParsedBasis { shells: ... }` — only shells matching `symbol` are kept.
    pub fn parse_nwchem(text: &str, symbol: &str, source: &str) -> Result<ParsedBasis, BasisLoadError> {
        let symbol_upper = symbol.to_ascii_uppercase();
        let mut shells: Vec<ShellSpec> = Vec::new();

        let lines: Vec<&str> = text.lines().collect();
        let mut i = 0;
        while i < lines.len() {
            let line = strip_comment(lines[i]).trim();
            i += 1;
            // Skip empty / BASIS / END / SET headers.
            if line.is_empty() || line.starts_with("BASIS")
                || line.starts_with("END") || line.starts_with("#") {
                continue;
            }

            // Element-symbol + angular-momentum-letter line: e.g. "H    S" or "C    S P"
            let tokens: Vec<&str> = line.split_whitespace().collect();
            if tokens.len() < 2 { continue; }   // not a header line; skip
            let elem = tokens[0].to_ascii_uppercase();
            // Strip suffix digits in element symbol (ghost-atom convention)
            let elem_alpha: String = elem.chars().take_while(|c| c.is_alphabetic()).collect();
            if elem_alpha != symbol_upper { 
                // Different element block; skip primitives until next header.
                while i < lines.len() {
                    let l = strip_comment(lines[i]).trim();
                    if l.is_empty() || is_block_header(l) { break; }
                    i += 1;
                }
                continue;
            }

            // The remaining tokens are angular-momentum letters (e.g., "S", "S P", "D").
            let l_letters: Vec<u8> = tokens[1..].iter()
                .filter_map(|t| ang_mom_from_letter(t))
                .collect();
            if l_letters.is_empty() {
                return Err(BasisLoadError::Parse {
                    file: source.to_string(),
                    line: i,
                    reason: format!("no angular-momentum letters in header line: {}", line),
                });
            }

            // Read primitive rows until blank line / next header / END.
            let n_columns = 1 + l_letters.len();   // exponent + one coeff per ang-mom
            let mut prim_rows: Vec<Vec<f64>> = Vec::new();
            while i < lines.len() {
                let l = strip_comment(lines[i]).trim();
                if l.is_empty() || is_block_header(l) || l.starts_with("END") { break; }
                let cols: Vec<f64> = l.split_whitespace()
                    .map(|s| s.parse::<f64>())
                    .collect::<Result<_, _>>()
                    .map_err(|_| BasisLoadError::Parse {
                        file: source.to_string(),
                        line: i + 1,
                        reason: format!("non-numeric primitive row: {}", l),
                    })?;
                if cols.len() != n_columns {
                    return Err(BasisLoadError::Parse {
                        file: source.to_string(),
                        line: i + 1,
                        reason: format!("expected {} columns, got {}: {}", n_columns, cols.len(), l),
                    });
                }
                prim_rows.push(cols);
                i += 1;
            }

            if prim_rows.is_empty() {
                return Err(BasisLoadError::Parse {
                    file: source.to_string(),
                    line: i,
                    reason: format!("empty primitive list for {} {:?}", symbol, l_letters),
                });
            }

            // Build one ShellSpec per angular-momentum letter (S P → 2 shells sharing exponents).
            for (l_idx, &l) in l_letters.iter().enumerate() {
                let exps: Vec<f64> = prim_rows.iter().map(|r| r[0]).collect();
                let coeffs: Vec<f64> = prim_rows.iter().map(|r| r[1 + l_idx]).collect();
                shells.push(ShellSpec {
                    l,
                    exponents: exps,
                    coeffs: vec![coeffs],   // single contraction column for this shell
                });
            }
        }

        Ok(ParsedBasis { shells })
    }

    /// Strip "# comment" suffix from a line.
    fn strip_comment(line: &str) -> &str {
        match line.find('#') { Some(i) => &line[..i], None => line }
    }

    /// Check whether a line starts a new block (element-symbol-letter line).
    fn is_block_header(line: &str) -> bool {
        let toks: Vec<&str> = line.split_whitespace().collect();
        if toks.len() < 2 { return false; }
        // First token is alphabetic (element); second is angular-momentum letter.
        let first_alpha = toks[0].chars().next().is_some_and(|c| c.is_alphabetic());
        let second_is_l = ang_mom_from_letter(toks[1]).is_some();
        first_alpha && second_is_l
    }

    /// Parse an angular-momentum letter to its integer l. SP / SPDFGHIK supported.
    fn ang_mom_from_letter(letter: &str) -> Option<u8> {
        match letter.to_ascii_uppercase().as_str() {
            "S" => Some(0), "P" => Some(1), "D" => Some(2), "F" => Some(3),
            "G" => Some(4), "H" => Some(5), "I" => Some(6), "K" => Some(7),
            _ => None,
        }
    }
    ```

    ```rust
    //! NWChem ECP-block parser (loading half of GTO-05).
    //!
    //! Source: pyscf/gto/basis/parse_nwchem_ecp.py (Apache-2.0).
    //!
    //! Format:
    //!   ECP
    //!   <Symbol> nelec <n_core>
    //!   <Symbol> ul              # local channel (l = -1 in upstream convention)
    //!   <n_power> <exponent> <coeff>
    //!   ...
    //!   <Symbol> <l-letter>      # projector channel (l = 0..6)
    //!   <n_power> <exponent> <coeff>
    //!   ...
    //!   END

    use pyscf_core::{EcpLoadError, ParsedEcp, EcpShell};

    pub fn parse_nwchem_ecp(text: &str, symbol: &str, source: &str) -> Result<ParsedEcp, EcpLoadError> {
        let symbol_upper = symbol.to_ascii_uppercase();
        let mut n_core: u32 = 0;
        let mut channels: Vec<EcpShell> = Vec::new();
        let mut current: Option<EcpShell> = None;

        for (i, raw_line) in text.lines().enumerate() {
            let line = match raw_line.find('#') { Some(j) => &raw_line[..j], None => raw_line };
            let line = line.trim();
            if line.is_empty() || line == "ECP" || line == "END" { continue; }

            let toks: Vec<&str> = line.split_whitespace().collect();
            // "Symbol nelec N" — sets core count.
            if toks.len() == 3 && toks[1].eq_ignore_ascii_case("nelec")
                && toks[0].eq_ignore_ascii_case(&symbol_upper) {
                n_core = toks[2].parse().map_err(|_| EcpLoadError::Parse {
                    file: source.into(), line: i + 1,
                    reason: format!("non-numeric n_core: {}", line),
                })?;
                continue;
            }
            // "Symbol ul" or "Symbol s/p/d/..." — opens a new channel.
            if toks.len() == 2 && toks[0].eq_ignore_ascii_case(&symbol_upper) {
                if let Some(c) = current.take() { channels.push(c); }
                let l: i8 = match toks[1].to_ascii_lowercase().as_str() {
                    "ul" => -1,
                    "s" => 0, "p" => 1, "d" => 2, "f" => 3, "g" => 4, "h" => 5, "i" => 6,
                    other => return Err(EcpLoadError::Parse {
                        file: source.into(), line: i + 1,
                        reason: format!("unknown ECP channel '{}'", other),
                    }),
                };
                current = Some(EcpShell { l, n_powers: vec![], exponents: vec![], coeffs: vec![] });
                continue;
            }
            // Primitive row: "<n_power> <exponent> <coeff>"
            if toks.len() == 3 {
                if let Some(ref mut c) = current {
                    let np: i32 = toks[0].parse().map_err(|_| EcpLoadError::Parse {
                        file: source.into(), line: i + 1, reason: format!("bad n_power: {}", line),
                    })?;
                    let exp: f64 = toks[1].parse().map_err(|_| EcpLoadError::Parse {
                        file: source.into(), line: i + 1, reason: format!("bad exp: {}", line),
                    })?;
                    let coef: f64 = toks[2].parse().map_err(|_| EcpLoadError::Parse {
                        file: source.into(), line: i + 1, reason: format!("bad coef: {}", line),
                    })?;
                    c.n_powers.push(np);
                    c.exponents.push(exp);
                    c.coeffs.push(coef);
                } else {
                    return Err(EcpLoadError::Parse {
                        file: source.into(), line: i + 1,
                        reason: "primitive row before any channel header".into(),
                    });
                }
            }
        }

        if let Some(c) = current.take() { channels.push(c); }
        Ok(ParsedEcp { n_core, channels })
    }
    ```

    ```rust
    //! GTO-02: format_basis — collapse 11 upstream input forms to 5 dispatch arms.
    //!
    //! Source: pyscf/gto/mole.py:418-466 + pyscf/gto/basis/__init__.py:621-728.
    //! Per Pitfall 4 (RESEARCH.md): group-by-symbol in first-occurrence order.

    use crate::basis;
    use crate::types::BasisInput;
    use pyscf_core::{BasisLoadError, ParsedAtom, ParsedBasis, PyscfRsError};
    use std::collections::HashMap;

    /// Resolve the user-supplied BasisInput to a per-element-symbol map.
    /// Output keys are UPPERCASE symbols (matches load_basis convention).
    /// Per-symbol order: first-occurrence in `atoms` (Pitfall 4).
    pub fn format_basis(
        input: &BasisInput,
        atoms: &[ParsedAtom],
    ) -> Result<HashMap<String, ParsedBasis>, PyscfRsError> {
        // Compute the unique-symbol list in first-occurrence order.
        let mut seen: Vec<String> = Vec::new();
        for (s, _) in atoms {
            // Normalise: strip ghost-atom suffix when looking up basis.
            let alpha: String = s.chars().take_while(|c| c.is_alphabetic()).collect();
            let upper = alpha.to_ascii_uppercase();
            if !seen.contains(&upper) { seen.push(upper); }
        }

        let mut out: HashMap<String, ParsedBasis> = HashMap::new();
        for symbol in &seen {
            let parsed = resolve_for_symbol(input, symbol)?;
            out.insert(symbol.clone(), parsed);
        }
        Ok(out)
    }

    fn resolve_for_symbol(input: &BasisInput, symbol: &str) -> Result<ParsedBasis, PyscfRsError> {
        match input {
            BasisInput::Name(name) => {
                basis::load_basis(name, symbol).map_err(PyscfRsError::from)
            }
            BasisInput::PerElement(map) => {
                // Look up the symbol in the user-supplied map; fall back to the empty BasisInput
                // (Phase 2: error if missing — upstream allows fallback to "default" key).
                let sub = map.get(symbol)
                    .or_else(|| map.get(&symbol.to_ascii_lowercase()))
                    .or_else(|| map.get("default"))
                    .ok_or_else(|| PyscfRsError::from(BasisLoadError::UnknownName {
                        name: format!("per-element basis missing for symbol '{}'", symbol),
                    }))?;
                resolve_for_symbol(sub, symbol)
            }
            BasisInput::NwchemText(text) => {
                basis::parse(text, symbol).map_err(PyscfRsError::from)
            }
            BasisInput::Cp2kText(text) => {
                basis::cp2k::parse_cp2k(text, symbol, "<cp2k-text>").map_err(PyscfRsError::from)
            }
            BasisInput::Parsed(parsed) => Ok(parsed.clone()),
        }
    }
    ```
